\section{Model and exact ideal toll}
All intervals below are half open.  A dyadic cell of depth $r$ has width
$2^{-r}$ and is represented by its midpoint.  Empty cells cost zero.  A
singleton not containing the target costs one, while the singleton containing
the target costs zero.  A nonsingleton containing the target is opened.  A
disjoint nonsingleton of width $w$ and midpoint distance $D$ is accepted, at
cost one, exactly when $w/D\leq\theta$; otherwise its two children are
visited, including an empty child.  Thus equality is accepted and one-child
chains are retained.

Let $0<\theta<1$, $\alpha=\theta^{-1}$, $b=\alpha-1/2$, and
$z=|2x-1|$.  Put $J=\lceil b\rceil$ and
$\ell_j=\lceil\log_2(j+1)\rceil$ for $0\leq j<J$.

\begin{proposition}[Exact filled-half toll]
\begin{equation}
 F_\theta(x)=1+\sum_{j=0}^{J-1}\sum_{d\geq\ell_j}
 \1\{z<(b-j)2^{-d}\}.                                      \tag{1.1}
\end{equation}
It belongs to $L^p(0,1)$ for every finite $p$ and
\begin{equation}\int_0^1F_\theta(x)\,dx=2/\theta.             \tag{1.2}\end{equation}
\end{proposition}
\begin{proof}
Reflection permits $x<1/2$.  At local depth $d$, offset $j$ in the opposite
half has
\[
 D/w=j+\tfrac12+2^dz.
\]
It is opened precisely when this is $<\alpha$, because equality is accepted.
This is the strict inequality in (1.1).  Such a cell exists precisely when
$j<2^d$, or $d\geq\ell_j$.  Opening one frontier cell replaces it by two and
therefore adds one leaf; (1.1) follows.

We now evaluate the mean without a geometric ``row telescoping'' shorthand.
For $a>0$ and an integer $\ell$ satisfying $a\leq2^\ell$, set
\[
 G(a,\ell)=\sum_{d\geq\ell}a2^{-d}=2a2^{-\ell}.
\]
The choice $\ell_j$ need not satisfy $b-j\leq2^{\ell_j}$, so define
$r_j=\max\{\ell_j,\lceil\log_2(b-j)\rceil\}$.  The terms
$\ell_j\leq d<r_j$ have integral one (with equality at the endpoint changing
only a null set), whereas the remaining terms have integral $(b-j)2^{-d}$.
Consequently
\begin{equation}
 \int F_\theta=1+\sum_{j<J}\{r_j-\ell_j+2(b-j)2^{-r_j}\}.       \tag{1.3}
\end{equation}
We prove the arithmetic identity rather than appeal to row telescoping.  The
following finite-tree observation will also be used in Section~3.  Regard
$0,\ldots,J-1$ as integers in the $2$-adic residue tree: a node at depth
$r$ is a nonempty set
\[
 N(r,a)=\{j:0\leq j<J,\ j\equiv a\pmod {2^r}\},
 \qquad 0\leq a<2^r.
\]
The root is $N(0,0)$, and a node with at least two elements has as children
its nonempty intersections with the two residue classes $a$ and $a+2^r$
modulo $2^{r+1}$.  Stop at a singleton.  This is a finite full binary tree:
every non-singleton node has exactly two nonempty children: explicitly,
$N(r,a)=\{a+k2^r:k_{\min}\leq k\leq k_{\max}\}$ after restricting
the displayed terms to $[0,J-1]$, and if it is not a singleton then both
$k_{\min}$ and $k_{\min}+1$ occur and have opposite parity.  Hence both
child residues occur.  At depth $r$ with $2^r\geq J$ every node is a
singleton.
The singleton containing $j$ first occurs at the least $r$ for which neither
$j-2^r$ nor $j+2^r$ lies in $[0,J-1]$, namely
\[
 2^r\geq j+1,\qquad 2^r>J-1-j.
\]
Because $J-1<b\leq J$, the second integer inequality is equivalent to
$2^r\geq b-j$.  Thus its depth is exactly
$r_j=\max\{\lceil\log_2(j+1)\rceil,
\lceil\log_2(b-j)\rceil\}$.  (The equivalence includes equality when
$b$ is an integer.)  Summing $2^{-r}$ down a finite full binary tree, each
split replacing $2^{-r}$ by two copies of $2^{-r-1}$, gives the Kraft identity
\begin{equation}
             \sum_{j<J}2^{-r_j}=1.                            \tag{1.4}
\end{equation}

Set $S(b)=\sum_{j<J}\{r_j-\ell_j+2(b-j)2^{-r_j}\}$.  Away
from the finitely many integer and dyadic breakpoints in any compact
$b$-interval, $J$ and every $r_j$ are constant, and (1.4) gives
$S'(b)=2\sum_j2^{-r_j}=2$.  There is no jump at a dyadic breakpoint:
when $y=b-j$ crosses $2^r$, the changing summand has limiting values
$0+2y2^{-r}=2$ from below and $1+2y2^{-r-1}=2$ from above.  Nor is there a
jump at an integer $b=N$: immediately to its right the new index $j=N$ has
$r_j=\ell_j$ and its summand tends to zero.  Finally, for
$1/2<b\leq1$ there is only $j=0$, $r_0=\ell_0=0$, and $S(b)=2b$.
Continuity across all breakpoints and the constant derivative between them
therefore prove, for every $b>1/2$,
\begin{equation}
 \sum_{j<J}\{r_j-\ell_j+2(b-j)2^{-r_j}\}=2b.              \tag{1.5}
\end{equation}
Substitution in (1.3) yields
$\int_0^1F_\theta=1+2b=2\alpha=2/\theta$, including all equality
cases (strict versus weak indicator endpoints are null).  Finally, there are
$J$ rows and each has at most $C+\log_+(1/z)$ nonzero terms, giving
$F_\theta\leq C_\theta(1+\log_+(1/z))$ and every finite moment.
\end{proof}

\section{Explicit Perron--Gordin algebra}
Let $T x=2x\pmod1$ and
\[
 Pq(y)=\tfrac12\{q(y/2)+q((y+1)/2)\}.
\]
For $0<c\leq1$ define
\[
 R_c(x)=\1_{x<c}+\1_{x>1-c},\qquad \bar R_c=R_c-2c,
\]
where overlap is counted twice and endpoint values are immaterial in $L^p$.
For the $r_j$ above set $c_j=(b-j)2^{-r_j}$ and
\[
 A=\sum_{j<J}\sum_{d\geq r_j}R_{(b-j)2^{-d}},\quad
 H^0=\sum_{j<J}\bar R_{c_j}.
\]
The series defining $A$ is pointwise finite off $\{0,1\}$ and is bounded by
$C_\theta(1+\log_+(1/x)+\log_+(1/(1-x)))$; hence $A\in L^p$ for finite $p$.

\begin{lemma}[Row identity]
In every finite $L^p$,
\begin{equation}F_\theta-\E F_\theta=H^0+A\circ T-A.            \tag{2.1}\end{equation}
\end{lemma}
\begin{proof}
For $0<c\leq1$ direct division into the two inverse branches gives
\begin{equation}R_c\circ T=\1_{\{|2x-1|<c\}}+R_{c/2}           \tag{2.2}\end{equation}
a.e.  Apply (2.2) with $c=(b-j)2^{-d}$ and sum first over
$r_j\leq d\leq N$.  The $R_{c/2}$ at depth $d$ cancels the $-R_c$ at depth
$d+1$, leaving $R_{c_j}\circ T-R_{(b-j)2^{-N-1}}$.  The last term converges
to zero in $L^p$ after centering (its support has measure $O(2^{-N})$).
The finitely many clipped depths $\ell_j\leq d<r_j$ are constants a.e.;
they disappear on centering.  Summing the finitely many rows proves (2.1).
\end{proof}

A second inverse-branch calculation gives
\begin{equation}P\bar R_c=\tfrac12\bar R_{\{2c\}}.             \tag{2.3}\end{equation}
Define the uniformly convergent series
\[
 k=\sum_{j<J}\sum_{s\geq1}2^{-s}\bar R_{\{2^sc_j\}},\qquad
 d=H^0+k-k\circ T,
\]
and set
\begin{equation}h=(A-\E A)+k.                                  \tag{2.4}\end{equation}
The bound $|\bar R_c|\leq2$ proves uniform convergence of $k$ and
$h\in L^p$ for every finite $p$.  Equations (2.1) and (2.4) give, a.e. and
in every finite $L^p$,
\begin{equation}F_\theta-\E F_\theta=d+h\circ T-h.             \tag{2.5}\end{equation}
Moreover, $P(q\circ T)=q$ and (2.3) imply $Pk=\sum_{j,s\geq1}
2^{-s-1}\bar R_{\{2^{s+1}c_j\}}$ and direct index shifting gives $Pd=0$.

\begin{lemma}[Canonical uniqueness]
If $f=d+Uh-h=d'+Uh'-h'$ in $L^2$, $Pd=Pd'=0$, and the only $L^2$ fixed
points of $P$ are constants, then $d=d'$ in $L^2$.
\end{lemma}
\begin{proof}
Put $q=h-h'$.  Then $d'-d=Uq-q$.  Applying $P$ gives $0=q-Pq$, so $q$ is a
$P$-fixed point and is constant.  Hence $Uq-q=0$.
\end{proof}
(The fixed-point assertion follows by Fourier coefficients:
$\widehat{Pq}(m)=\widehat q(2m)$; iteration and square summability force every
nonzero coefficient to vanish.)

\section{Fourier reduction, resonance, and cusp}
For positive odd $u$ put
\[
 \Psi_u(t)=\sum_{s\geq0}2^{-s}\sin(2\pi2^sut).
\]
The Fourier calculation from (2.3) gives zero even coefficients and
\begin{equation}
 \widehat d(u)={1\over\pi u}\sum_{j<J}\Psi_u(c_j),\qquad u>0\text{ odd}. \tag{3.1}
\end{equation}
The splitting identity
\[
 \Psi_u(t/2)+\Psi_u((t+1)/2)=\Psi_u(t)                          \tag{3.2}
\]
is obtained by cancellation of the two level-zero sines and reindexing both
tails.

\begin{lemma}[Endpoint collapse]
For every positive odd integer $u$,
\begin{equation}\sum_{j<J}\Psi_u(c_j)=\Psi_u(\{b\}).           \tag{3.3}\end{equation}
\end{lemma}
\begin{proof}
We give the complete inverse-doubling prefix tree.  Its combinatorial nodes
are the nonempty residue sets $N(r,a)$ defined in the proof of (1.4).  Attach
to such a node the circle label
\[
 L(r,a)=\left\{\frac{b-a}{2^r}\right\}\in\mathbb R/\mathbb Z.
\]
The parent map sends each node at depth $r+1$ to the unique residue node at
depth $r$ containing it.  If $N(r,a)$ is not a singleton, its two children
are $N(r+1,a)$ and $N(r+1,a+2^r)$, after choosing $0\leq a<2^r$.
Their labels, in one order, are
\[
          L(r,a)/2,\qquad (L(r,a)+1)/2
          \quad\pmod 1.
\]
Indeed the raw quotients $(b-a)/2^{r+1}$ and
$(b-a-2^r)/2^{r+1}$ differ from those two representatives only by integers
and by interchanging the two.  Thus (3.2), together with the
one-periodicity of $\Psi_u$, says that the value at every internal node is
the sum of the values at its two children.

The stopping rule is: stop precisely when $N(r,a)$ is a singleton.  The
argument in Section~1 proves both finiteness (all nodes have stopped once
$2^r\geq J$) and that the exact leaves, with multiplicity as nodes, are
\[
       N(r_j,j\bmod 2^{r_j})=\{j\},\qquad 0\leq j<J.
\]
The label of this leaf is
$\{(b-j)/2^{r_j}\}=\{c_j\}$.  Here $0<c_j\leq1$: if $c_j=1$
we use its circle representative $0$, which changes neither $\Psi_u$ nor
the identity.  Distinct indices remain distinct leaves even if numerical
circle labels happen to agree; hence every multiplicity in the displayed
sum is retained.  Iterating the child identity over this finite tree makes
the root value equal the sum of its leaf values.  The root label is
$L(0,0)=\{b\}$, proving (3.3).  The only equality cases in constructing
the leaves are $2^{r_j}=j+1$ or $2^{r_j}=b-j$; the singleton criterion above
uses respectively the exact conditions $2^{r_j}>j$ and
$2^{r_j}>J-1-j$, so both cases are already included.
\end{proof}

\begin{lemma}[Simultaneous odd-chain lemma]\label{lem:simultaneous-odd}
For $t\in[0,1)$, $\Psi_u(t)=0$ for every positive odd $u$ iff
$t\in\{0,1/2\}$.
\end{lemma}
\begin{proof}
The reverse direction is termwise.  (This is a simultaneous assertion; no scalar zero-set claim is used.)  Since $\Psi_u(t)=\Psi_1(ut)$, if $t$ is
irrational then its odd multiples are dense modulo one (they form an orbit
under irrational rotation by $2t$).  Continuity would force $\Psi_1$ to
vanish identically, contradicting
$\int_0^1\Psi_1(x)\sin(2\pi x)\,dx=1/2$; uniform convergence justifies
termwise integration.

Let $t=p/q$ be reduced.  If $q$ is odd, every residue modulo $q$ has an odd
integer representative.  Vanishing for all odd $u$ therefore makes the
sampled vector $(\Psi_1(a/q))_{a\bmod q}$ zero.  Absolute convergence and
\[
 \sum_{a=0}^{q-1}\sin(2\pi ka/q)\sin(2\pi a/q)
 ={q\over2}({\bf1}_{k\equiv1\ (q)}-{\bf1}_{k\equiv-1\ (q)})
\]
(valid since $q>2$) show that $2/q$ times its frequency-one sine
coefficient equals
\[
 C_1=\sum_{s:2^s\equiv1\ (q)}2^{-s}
     -\sum_{s:2^s\equiv-1\ (q)}2^{-s}.
\]
The first sum contains the term $s=0$ of mass one.  The second is a proper
subseries of $\sum_{s\geq1}2^{-s}=1$: two consecutive powers cannot both be
$-1$ modulo an odd $q>1$.  Thus $C_1>0$, a contradiction.  The case $q=1$
is $t=0$.

If $q=2^av$ with $v$ odd, choose the allowed odd multiplier $u=v$, reducing
to $p/2^a$.  For $a\geq2$, let $\zeta$ be a primitive $2^a$-th root.  The
finite equality $\Psi_1(p/2^a)=0$ would put
$\zeta^p-\zeta^{-p}$ in $\mathbb Q(\zeta^2)$: explicitly, after multiplying
by $2i$ and using that the terms $s\geq a$ vanish,
\[
 \zeta^p-\zeta^{-p}
 =-\sum_{s=1}^{a-1}2^{-s}
       (\zeta^{2^sp}-\zeta^{-2^sp})\in\mathbb Q(\zeta^2).
\]
The automorphism $\zeta\mapsto-\zeta$ fixes that
subfield but negates this nonzero element, a contradiction.  For $a=1$ the
reduced fraction is $1/2$.
\end{proof}

Equations (3.1), (3.3), Lemma~\ref{lem:simultaneous-odd}, and Parseval give
\[
 d=0\iff\{b\}\in\{0,1/2\}\iff b\in\tfrac12\mathbb Z.
\]
Since $b=\theta^{-1}-1/2$ and $0<\theta<1$, this is equivalent to
$2/\theta=2b+1=:Q$ with an integer $Q\ge3$, and conversely.  Thus
\[
 c(\theta):=\|d_\theta\|_2^2=0\iff\theta=2/Q,\quad Q\ge3.       \tag{3.4}
\]
Parseval (including negative frequencies) and the odd cosine series give
\[
 c(t)={2\over\pi^2}\sum_{\substack{u\ge1\\u\ {\rm odd}}}{\Psi_u(t)^2\over u^2}
 ={1\over2}\sum_{r,s\ge0}2^{-r-s}
 \{\|(2^r+2^s)t\|_{\mathbb Z}-\|(2^r-2^s)t\|_{\mathbb Z}\}.    \tag{3.5}
\]
Fubini is justified by
$\sum_{r,s}2^{-r-s}\sum_{u\ {\rm odd}}u^{-2}<\infty$; the kernel series is
uniformly absolutely convergent and hence continuous.

\begin{proposition}[Exact resonance cusps]\label{prop:exact-cusps}
As $\varepsilon\to0$,
\[
 c(\varepsilon)=6|\varepsilon|+o(|\varepsilon|),\qquad
 c(1/2+\varepsilon)=2|\varepsilon|+o(|\varepsilon|).           \tag{3.6}
\]
At $\theta_0=2/Q$ the corresponding slopes are $3Q^2/2$ for odd $Q$ and
$Q^2/2$ for even $Q$.
\end{proposition}
\begin{proof}
Put $\phi(x)=\|x\|_{\mathbb Z}$.  For fixed $(r,s)$ the summand in (3.5),
divided by $|\varepsilon|$, tends to $2^{-\max(r,s)}$.  The Lipschitz
property of $\phi$, with the smaller power viewed as the perturbation, gives
\[
 {2^{-r-s-1}\over|\varepsilon|}
 |\phi((2^r+2^s)\varepsilon)-\phi((2^r-2^s)\varepsilon)|
 \le2^{-\max(r,s)}.
\]
Since $\sum_{r,s}2^{-\max(r,s)}=\sum_{j\ge0}(2j+1)2^{-j}=6$, dominated
convergence proves the integer profile on both sides.

For $t=1/2+\varepsilon$, split (3.5) into $(0,0)$, pairs with exactly one
zero index, and pairs with both indices positive.  The first contribution is
$|\varepsilon|$.  Combining the two ordered mixed pairs at level $j\ge1$
and using $\phi(1/2+x)=1/2-\phi(x)$ gives exactly
\[
 B(\varepsilon)=\sum_{j\ge1}2^{-j}
 \{\phi((2^j-1)\varepsilon)-\phi((2^j+1)\varepsilon)\}.
\]
For fixed $j$ the brace divided by $|\varepsilon|$ tends to $-2$.
Its normalized absolute value is at most $2$, since its arguments differ by
$2\varepsilon$; dominated convergence gives
$B(\varepsilon)=-2|\varepsilon|+o(|\varepsilon|)$.  Removing one power of
two from each positive index shows the last part is exactly
$\frac14c(2\varepsilon)=3|\varepsilon|+o(|\varepsilon|)$.  Summing proves
the half-integer profile.  Finally
$b-(Q-1)/2=-(Q^2/4)(\theta-2/Q)+O((\theta-2/Q)^2)$.
The base point is integral for odd $Q$ and half-integral for even $Q$,
which proves the stated positive (and hence genuinely cuspidal) slopes.
\end{proof}

\section{Strict stopping and exact martingale variance}
For iid fair strings $X_i$, let $\tau_i$ be the least depth at which label
$i$ is alone and put
\[
 M_n=\sum_i\sum_{0\leq r<\tau_i}d(T^rX_i),\quad c=\norm d_2^2.
\]
Split by the first bit.  Let $B\sim\mathrm{Bin}(n,1/2)$ and, conditional on
$B=k$, let the two rescaled suffix samples be independent canonical samples
of sizes $k,n-k$.  Write $D_a(k)=\sum_{i=1}^kd((a+Y_i)/2)$ and let
$M_a(k)$ be the strict descendant sum in that branch.  Then pathwise
\[
 M_n=D_0(B)+M_0(B)+D_1(n-B)+M_1(n-B).
\]
Conditional branch independence and zero descendant means eliminate cross-branch
covariances involving descendants.  The unconditional root sum has variance
$nc$: before conditioning on $B$ it is exactly $\sum_{i=1}^n d(X_i)$.
For clarity, it is generally false that its conditional variance given $B$ is
$nc$.  If $\mu_0=\int_0^1d(y/2)\,dy$, then $Pd=0$ gives branch means
$\mu_0,-\mu_0$; conditional variance contributes $n(c-\mu_0^2)$ and
variance of the conditional mean $(2B-n)\mu_0$ contributes $n\mu_0^2$.
Their sum is $nc$.  Define
$p_a(k)=\Cov(D_a(k),M_a(k))$.  The same identity gives $D_1(k)=-D_0(k)$ under
the common suffix coupling, while descendant laws agree, hence
$p_1(k)=-p_0(k)$.  Total variance therefore gives
\begin{equation}v_n=2\E v_B+nc+2\E\{p_0(B)+p_1(n-B)\}=2\E v_B+nc,\tag{4.1}\end{equation}
because $B\stackrel d=n-B$.  With $v_0=v_1=0$, comparison with the external
path recurrence proves
\begin{equation}\Var(M_n)=cnS_{n-1},\quad
 S_m=\sum_{r\geq0}[1-(1-2^{-r})^m]=\log_2m+O(1).                \tag{4.2}\end{equation}

\section{Finite Walsh martingale and CLT}\label{sec:walsh-clt}
We now prove the finite-Walsh assertion, including the bracket concentration
which will be used below.  Write
\(
 \mathcal D_m=\sigma(B_1,\ldots,B_m)
\)
for the dyadic sigma field on one infinite fair bit string and define, as an
$L^2$ conditional expectation,
\begin{equation}
 d^{(m)}=\E(d\mid\mathcal D_m).
 \tag{5.1}
\end{equation}
Put $\epsilon_s=(-1)^{B_s}$.  The Walsh basis of $L^2(\mathcal D_m)$ gives
uniquely
\begin{equation}
 d^{(m)}=\sum_{S\subseteq\{1,\ldots,m\}}a_S\chi_S,
 \qquad \chi_S=\prod_{s\in S}\epsilon_s,
 \qquad a_S=\E(d\chi_S).                                    \tag{5.2}
\end{equation}
The empty coefficient is zero because $\E d=0$.  Moreover $Pd=0$ says that
$\E(d\mid B_2,B_3,\ldots)=0$.  Taking conditional expectations shows that
the coefficient of every $S$ not containing $1$ is zero.  Thus (5.2) is a
sum over $1\in S$; in particular $j(S):=\max S$ is defined.  The same
observation, or a direct average over the first bit, proves $Pd^{(m)}=0$.

For label $i$ write $\epsilon_{i,s}=(-1)^{B_{i,s}}$.  The translate of
$\chi_S$ based at trie depth $r$ is
\begin{equation}
 \chi_{i,r,S}=\prod_{s\in S}\epsilon_{i,r+s}.                \tag{5.3}
\end{equation}
Consequently the finite approximation to the strict-stopped sum is
\begin{equation}
 M_n^{(m)}=\sum_{i=1}^n\sum_{r\ge0}{\bf1}_{\{\tau_i>r\}}
                   \sum_{S\ni1}a_S\chi_{i,r,S}.              \tag{5.4}
\end{equation}
Every sum in (5.4) is almost surely finite.  Reveal all labels one bit-column
at a time and set
\begin{equation}
 \mathcal F_t=\sigma(\epsilon_{i,s}:1\le i\le n,1\le s\le t),
 \quad t\ge0,\qquad \mathcal F_0=\{\varnothing,\Omega\}.      \tag{5.5}
\end{equation}
Assign the summand $(i,r,S)$ in (5.4) to the unique bit time
$t=r+j(S)$.  Factoring out its last bit gives
\begin{align}
 M_n^{(m)}&=\sum_{t\ge1}\Delta_{n,t},
 &\Delta_{n,t}&=\sum_{i=1}^n\epsilon_{i,t}C_{i,t-1},          \tag{5.6}\\
 C_{i,t-1}&=\sum_{\substack{S\ni1\\j(S)\le\min(m,t)}}a_S
 {\mathbf 1}_{\{\tau_i>t-j(S)\}}
 \prod_{s\in S\setminus\{j(S)\}}
       \epsilon_{i,t-j(S)+s}.                                \tag{5.7}
\end{align}
Isolation at depth $r$ is decided by the first $r$ bits of every label.
In (5.7), $r=t-j(S)\le t-1$, and every remaining bit index is at most
$t-1$.  Hence $C_{i,t-1}$ is $\mathcal F_{t-1}$-measurable.  Conditional on
$\mathcal F_{t-1}$ the $\epsilon_{i,t}$ are independent centered
Rademachers.  Thus (5.6) is a martingale-difference representation, and its
predictable bracket is
\begin{equation}
 Q_n=\sum_{t\ge1}B_t,
 \qquad B_t=\E(\Delta_{n,t}^2\mid\mathcal F_{t-1})
            =\sum_{i=1}^n C_{i,t-1}^2.                        \tag{5.8}
\end{equation}

We record all activity estimates needed below.  Put
$A_r=\sum_i{\mathbf 1}_{\{\tau_i>r\}}$ and $p_r=1-(1-2^{-r})^{n-1}$.
For a fixed label, strict activity at depth $r$ means that at least one of
the other $n-1$ labels has its length-$r$ prefix.  Therefore
\begin{equation}
 \E A_r=np_r.                                                 \tag{5.9}
\end{equation}
Also $A_r^2\le nA_r$, whence
\begin{equation}
 \sum_{r\ge0}\E A_r^2\le n^2\sum_{r\ge0}min(1,(n-1)2^{-r})
                         =O(n^2\log(n+1)).                    \tag{5.10}
\end{equation}
Here $1-(1-x)^{n-1}\le\min(1,(n-1)x)$, and splitting at
$\lceil\log_2(n\vee2)\rceil$ proves the last estimate, including the
geometric tail.  Finally, summing (5.9) and using the definition in (4.2),
\begin{equation}
 \sum_{r\ge0}\E A_r=nS_{n-1}.                                \tag{5.11}
\end{equation}
If $K_m=\sum_{S\ni1}|a_S|$, Cauchy--Schwarz in the finite sum of possible
maxima (or simply $(\sum_jx_j)^2\le m\sum_jx_j^2$) gives, with a constant
depending only on $m$ and the coefficients,
\begin{equation}
 B_t\le K_m^2\sum_{j=1}^m A_{t-j},                            \tag{5.12}
\end{equation}
where $A_r=0$ for $r<0$.  It follows from (5.10),
$(\sum_{j=1}^m A_{t-j})^2\le m\sum_jA_{t-j}^2$, that
\begin{equation}
 \sum_t\E B_t^2=O_m(n^2\log(n+1)).                           \tag{5.13}
\end{equation}

There are two exact ways to compute the mean bracket.  Martingale isometry
in (5.6) says $\E Q_n=\Var M_n^{(m)}$.  Since $Pd^{(m)}=0$, the proof of
(4.2), with $d^{(m)}$ in place of $d$, applies without alteration.  Parseval
in (5.2) therefore yields the exact identity
\begin{equation}
 \E Q_n=\|d^{(m)}\|_2^2 nS_{n-1}
       =\Big(\sum_{S\ni1}a_S^2\Big)nS_{n-1}.                 \tag{5.14}
\end{equation}

Here is the promised complete bounded-influence argument.  Replace $X_i$ by
an independent $X_i'$ and denote the resulting bracket by $Q_n^{(i)}$.
In the augmented collection
$\{X_1,\ldots,X_n,X_i'\}$ let
\begin{equation}
 H=1+\max_{x\ne y}\operatorname{lcp}(x,y),                   \tag{5.15}
\end{equation}
where $\operatorname{lcp}$ is common-prefix length.  At a fixed depth $r$,
only the old prefix cell of $X_i$ and the new prefix cell of $X_i'$ have
changed occupancy.  Apart from label $i$, an isolation indicator in the old
cell can change only if that label had $i$ as its sole companion; there is at
most one such label, since two would be companions of each other.  The same
argument gives at most one companion in the new cell.  Thus at each depth
the changed isolation indicators belong to label $i$ and at most two other
labels.  No indicator can change at depth $r\ge H$: by (5.15) all augmented
prefixes are then distinct.

For labels other than $i$, their bits do not change; hence a changed term in
(5.7) must contain one of those changed activity indicators.  A fixed pair
(label,depth $r$) occurs only at the times $t=r+j$, $1\le j\le m$, so the
number of affected $(i,t)$ entries of (5.8), counted with multiplicity, is
at most $2mH$ for the two possible companions.  For the replaced label,
both its old and new coefficient chains vanish after depth $H-1$; allowing
the $m$ possible terminal offsets, their comparison affects at most
$H+m$ time entries.  Every coefficient, before and after replacement, has absolute value at most
$K_m$.  Hence the absolute difference of their squares is at most $K_m^2$.
There are at most $2mH+H+m\le4m(H+m)$ affected entries.  Summing these
entrywise bounds gives the deterministic estimate
\begin{equation}
 |Q_n-Q_n^{(i)}|\le D_m(H+m),\qquad D_m=4mK_m^2.             \tag{5.16}
\end{equation}
This enumeration accounts both for the factor $m$ (the possible Walsh
maxima) and for every label whose isolation can change.

For completeness, a fixed pair of independent strings has common prefix at
least $h$ with probability $2^{-h}$.  A union bound over the
$\binom{n+1}{2}$ augmented pairs gives
\begin{equation}
 \Pp(H>h+1)\le\binom{n+1}{2}2^{-h}.                           \tag{5.17}
\end{equation}
Taking $h_0=\lceil2\log_2(n+1)\rceil$ and using
$\E Z^2=\sum_{k\ge0}(2k+1)\Pp(Z>k)$ for integer $Z\ge0$, the part below
$h_0$ is $O(h_0^2)$ and (5.17) makes the remaining geometric sum
$O(h_0+1)$.  Hence $\E(H+m)^2=O_m(\log^2(n+1))$.  The
Efron--Stein inequality and (5.16) now give
\begin{equation}
 \Var Q_n\le\frac12\sum_{i=1}^n\E(Q_n-Q_n^{(i)})^2
             =O_m(n\log^2(n+1)).                             \tag{5.18}
\end{equation}
In particular, if $\|d^{(m)}\|_2>0$, (5.14), (5.18), and
$S_{n-1}\sim\log_2n$ imply
\begin{equation}
 {Q_n\over \|d^{(m)}\|_2^2nS_{n-1}}\longrightarrow1
 \quad\hbox{in probability}.                                \tag{5.19}
\end{equation}

Conditionally on $\mathcal F_{t-1}$, the summands in (5.6) are independent
centered variables.  Expanding the fourth power (only index patterns $4$ and
$2+2$ survive) gives exactly
\begin{equation}
 \E(\Delta_{n,t}^4\mid\mathcal F_{t-1})
 =3B_t^2-2\sum_iC_{i,t-1}^4\le3B_t^2.                        \tag{5.20}
\end{equation}
Consequently (5.13) yields
\begin{equation}
 \sum_t\E\Delta_{n,t}^4=O_m(n^2\log(n+1))
                         =o((n\log n)^2).                    \tag{5.21}
\end{equation}
For every $\eta>0$, conditional Markov followed by expectation gives
\begin{align}
 \E\sum_t {\E[\Delta_{n,t}^2{\mathbf 1}_{\{|\Delta_{n,t}|>
 \eta\sqrt{nS_{n-1}}\}}\mid\mathcal F_{t-1}]\over nS_{n-1}}
 &\le {\sum_t\E\Delta_{n,t}^4\over
             \eta^2(nS_{n-1})^2}\longrightarrow0.           \tag{5.22}
\end{align}
Thus the conditional Lindeberg quantity tends to zero in $L^1$, hence in
probability.

To make the triangular array finite, take explicitly
\begin{equation}T_n=\lceil4\log_2(n+1)\rceil+m.               \tag{5.23}
\end{equation}
By martingale orthogonality, (5.12), and (5.9),
\begin{align}
 \E\Big|\sum_{t>T_n}\Delta_{n,t}\Big|^2
 &=\sum_{t>T_n}\E B_t
 \le K_m^2m\sum_{r>\lceil4\log_2(n+1)\rceil}\E A_r\\
 &\le K_m^2mn(n-1)\sum_{r>\lceil4\log_2(n+1)\rceil}2^{-r}
 =O_m(n^{-2}).                                                \tag{5.24}
\end{align}
It is therefore $o(1)$ in $L^2$ after division by $\sqrt{n\log n}$.

We use the following standard martingale triangular-array CLT in its precise
form: if a finite row $(Z_{n,t},\mathcal F_{n,t})$ consists of square
integrable martingale differences, if
$\sum_t\E(Z_{n,t}^2\mid\mathcal F_{n,t-1})\to1$ in probability, and for
every $\eta>0$,
$\sum_t\E(Z_{n,t}^2{\mathbf 1}_{|Z_{n,t}|>\eta}\mid
\mathcal F_{n,t-1})\to0$ in probability, then $\sum_tZ_{n,t}\Rightarrow
N(0,1)$.  Apply it to $1\le t\le T_n$ with
$Z_{n,t}=\Delta_{n,t}/(\|d^{(m)}\|_2\sqrt{nS_{n-1}})$.
The omitted bracket has expectation bounded by (5.24), so Markov's
inequality and (5.19) verify the bracket hypothesis; (5.22) verifies
Lindeberg.  Slutsky's theorem and (5.24) prove
\begin{equation}
 {M_n^{(m)}\over\|d^{(m)}\|_2\sqrt{nS_{n-1}}}
 \Rightarrow N(0,1).                                         \tag{5.25}
\end{equation}

Finally, the increasing dyadic sigma fields generate the Borel sigma field
modulo null sets.  The $L^2$ martingale convergence theorem applied to
$\E(d\mid\mathcal D_m)$ gives
\begin{equation}\|d-d^{(m)}\|_2\longrightarrow0.             \tag{5.26}\end{equation}
Both terms are killed by $P$, so (4.2), applied to their difference, gives
the exact estimate
\begin{equation}
 \E|M_n-M_n^{(m)}|^2=\|d-d^{(m)}\|_2^2nS_{n-1}.              \tag{5.27}
\end{equation}
If $c=\|d\|_2^2>0$, then $\|d^{(m)}\|_2^2\to c$ and is positive for all
large $m$.  For such a fixed $m$, (5.25) and
$S_{n-1}/\log_2n\to1$ give the normal limit with variance
$\|d^{(m)}\|_2^2/c$ under normalization $\sqrt{cn\log_2n}$; (5.27) makes
the $L^2$ distance to the corresponding normalization of $M_n$ at most
$\|d-d^{(m)}\|_2/\sqrt c+o(1)$.  Letting first $n\to\infty$ and then
$m\to\infty$ (the converging-together lemma) proves
\begin{equation}
 {M_n\over\sqrt{c(\theta)n\log_2n}}\Rightarrow N(0,1).
                                                                    \tag{5.28}
\end{equation}
\section{Endpoint error}
Finite pathwise telescoping of (2.5), with no optional-stopping claim, gives
\[
 V_{\theta,n}=(\E F_\theta)L_n+M_n+H_n,\qquad
 H_n=\sum_i\{h(T^{\tau_i}X_i)-h(X_i)\}.
\]
Let $\mathcal G_n$ be the sigma field generated by all labelled stopped
prefixes.  Prefix freeness and cylinder factorization imply that, conditional
on $\mathcal G_n$, the unused suffixes $T^{\tau_i}X_i$ are independent
uniform strings and independent of $\mathcal G_n$.  Therefore
\[
 \E[\Var(\sum_i h(T^{\tau_i}X_i)\mid\mathcal G_n)]=n\Var(h),
\]
and its conditional mean is $n\E h=0$.  The initial sum consists of iid
copies and also has variance $n\Var(h)$.  Cauchy--Schwarz controls their
covariance, proving
\begin{equation}\Var(H_n)\leq4n\Var(h)=O_\theta(n).             \tag{6.1}\end{equation}
Set $\mu=\E F_\theta=2/\theta$.  Centering the displayed pathwise identity
gives
\[
 V_{\theta,n}-\E V_{\theta,n}
 =M_n+R_n,\qquad
 R_n=\mu(L_n-\E L_n)+(H_n-\E H_n).
\]
The established path-length estimate $\Var(L_n)=O(n)$ and (6.1), together
with Cauchy--Schwarz, give $\Var(R_n)=O_\theta(n)$.  A second application of
Cauchy--Schwarz gives
$|\Cov(M_n,R_n)|\le\sqrt{c nS_{n-1}\,O_\theta(n)}
=o(n\log n)$.  Therefore (4.2) implies, whenever $c(\theta)>0$,
\[
 \Var(V_{\theta,n})\sim c(\theta)n\log_2n.
\]
Moreover $R_n/\sqrt{c(\theta)n\log_2n}\to0$ in $L^2$; (5.28) and Slutsky's
theorem prove the complete strict-stopped nonresonant limit
\[
 {V_{\theta,n}-\E V_{\theta,n}\over
       \sqrt{c(\theta)n\log_2n}}\Rightarrow N(0,1).
\]
Since the variance ratio tends to one, the same statement holds with the
true standard deviation $\sqrt{\Var(V_{\theta,n})}$ in the denominator.
No adaptive-discrepancy assertion is used in this strict-stopped theorem.

\section{The adaptive root defect and an arithmetic-free guard}
\label{sec:adaptive-guard-dossier}

Throughout this section $0<\theta<1$ is fixed, $\alpha=\theta^{-1}$, and
all cells are half open.  Equality in $w/D\leq\theta$ means acceptance.  Empty
cells cost zero, a source singleton different from the target costs one, and
a nonsingleton containing the target is opened, including along a one-child
chain.  These conventions are part of every identity below.

For a finite labelled configuration $S\subset[0,1)$ and $x\in S$, define the
frontier $\mathcal A_\theta(x,S;I)$ of a dyadic source cell $I$ recursively:
(i) it is empty if $S\cap I=\varnothing$; (ii) it is the singleton $I$ if
$S\cap I$ has one element different from $x$; (iii) if $I$ contains $x$, it
is the union of the frontiers of the children, with the leaf containing only
$x$ assigned cost zero; (iv) if $I$ is disjoint from $x$, occupied by at least
two points, and accepted, it is $\{I\}$; otherwise it is the union of the two
child frontiers.  The recursion terminates because distinct points have finite
separation depth.  The filled frontier is defined by the same geometric
recursion without empty or singleton stopping.

Let $Y_\theta(S)$ be the sum, over targets, of the two-direction adaptive
cross-root frontier cardinality, and let
\begin{equation}
 \Phi_\theta(S)=\sum_{x\in S}F_\theta(x),\qquad
 D_\theta(S)=Y_\theta(S)-\Phi_\theta(S).                 \label{eq:defect-def}
\end{equation}
For iid uniform samples of size $n$ abbreviate this by $D_{\theta,n}$, and put
$\bar D_{\theta,n}=D_{\theta,n}{\bf1}_{\{n\ge2\}}$.

\begin{lemma}[finite exterior frontier]\label{lem:finite-exterior}
Put
\begin{equation}
 M=M(\theta)=2^{\max\{2,\lceil\log_2\max(1,4/\theta-2)\rceil\}},\qquad h=\log_2M.
                                                               \label{eq:Mguard}
\end{equation}
Every filled exterior branch, other than the interactions between the two
middle quarters, terminates by depth $h$.
\end{lemma}
\begin{proof}
For a target in the left half, an opposite-half depth-$d$ cell at offset $j$
has
\begin{equation}D/w=j+\tfrac12+2^d|2x-1|.                    \label{eq:rowdistance}
\end{equation}
It is opened precisely when $2^d|2x-1|<\alpha-\tfrac12-j$; hence there are
finitely many rows.  At depth $h$, every exterior source cell has distance at
least $1/4+1/(2M)$.  Definition \eqref{eq:Mguard} gives
$M\ge4\alpha-2$, or
\begin{equation}1/4+1/(2M)\ge\alpha/M.                       \label{eq:guardineq-d}
\end{equation}
Thus that source cell is accepted, including equality.  If an ancestor was
accepted, the branch stopped earlier; otherwise descent reaches depth $h$.
The sole pairs not covered by the distance $1/4+1/(2M)$ are the two middle
quarters.  Reflection proves the right-target assertion.
\end{proof}

Let $C_j=[j/M,(j+1)/M)$, let
$\mathcal G_n=\{N(C_j)\ge1:0\le j<M\}$, let $J_*=[1/4,3/4)$,
and let $K_n=N(J_*)$.
\begin{theorem}[coefficient-one inward copy]\label{thm:inward-copy-dossier}
On $\mathcal G_n$,
\begin{equation}D_{\theta,n}=D^{\rm in}_{\theta,K_n}.          \label{eq:inward-copy-dossier}
\end{equation}
Conditional only on $K_n=k$, the right side has the canonical law of
$D_{\theta,k}$.  Moreover
\begin{equation}K_n\le n-M/2\quad\hbox{on }\mathcal G_n,\qquad
 \Pr(\mathcal G_n^c)\le M(1-1/M)^n.                          \label{eq:guardfail-d}
\end{equation}
There is exactly one copy and no boundary toll.
\end{theorem}
\begin{proof}
On $\mathcal G_n$ every cell at depth at most $h$ is occupied.  Before a
filled exterior branch terminates, the adaptive traversal can therefore
neither prune it empty nor stop at a singleton.  Lemma
\ref{lem:finite-exterior} says that it terminates by depth $h$.  At its last
node, an accepted occupied cell and an occupied singleton both contribute
one.  Hence filled and adaptive exterior contributions cancel target by
target in \eqref{eq:defect-def}.

The uncancelled directed interactions are precisely $[1/4,1/2)$ against
$[1/2,3/4)$ and the reverse direction.  The increasing map
$\sigma(x)=2x-1/2$ maps those quarters to the two root halves.  It doubles
both widths and midpoint distances, preserves $w/D$, equality, dyadic
containment, multiplicity, empty children, singleton stopping, clipping and
each link of a one-child chain.  The two directed interactions are therefore
the two directions of one canonical root problem, rather than two copies.
This proves \eqref{eq:inward-copy-dossier}.  Given $K_n=k$, the $k$ points in
$J_*$ are iid uniform there, so their $\sigma$-images are canonical.  This
conditioning is not conditioning on $\mathcal G_n$.  The outer $M/2$ cells
each contain a point on $\mathcal G_n$, proving the size loss; a union bound
proves \eqref{eq:guardfail-d}.
\end{proof}

\begin{proposition}[all root-defect moments]\label{prop:defect-moments-d}
For every finite $p>0$, $\sup_n\mathbb E|D_{\theta,n}|^p<\infty$.
\end{proposition}
\begin{proof}
The actual cross cost is at most $n(n-1)$.  From the exact toll formula,
$F_\theta(x)\le C_\theta(1+\log_+|2x-1|^{-1})$, whose every finite moment is
finite.  Minkowski therefore gives
$\mathbb E|D_{\theta,n}|^{2p}\le P_{\theta,p}(n)$ for a polynomial $P$.
By Cauchy--Schwarz and \eqref{eq:guardfail-d},
\begin{equation}e_n:=\mathbb E[|D_{\theta,n}|^p;\mathcal G_n^c]
 \le P_{\theta,p}(n)^{1/2}M^{1/2}(1-1/M)^{n/2},              \label{eq:en}
\end{equation}
so $\sum_ne_n<\infty$.  Put $A_n=\max_{0\le j\le n}\mathbb E|D_{\theta,j}|^p$.
On the good event use the pathwise copy and then enlarge the expectation to
the unconditional inward sample; this order avoids conditioning bias.  The
size loss yields
\begin{equation}
 \mathbb E|D_{\theta,n}|^p\le A_{n-M/2}+e_n\le A_{n-1}+e_n.
\end{equation}
Consequently $A_n\le A_{M/2}+\sum_{j>M/2}e_j$, uniformly in $n$.
\end{proof}

\section{Exact discrepancy recursion and linear variance}
Partition every ordered interaction by the least common dyadic ancestor of
its target and source.  The finite-tree partition is disjoint and exhaustive;
empty and singleton nodes contribute no internal toll, while one-child
internal nodes remain.  Thus, pathwise,
\begin{align}
 \Delta_{\theta,n}:=T_{\theta,n}-V_{\theta,n}
 &=\sum_{u:N_u\ge2}\bar D_\theta(S_u),                       \label{eq:finite-defect-sum}\\
 \Delta_{\theta,n}&=\Delta^{(0)}_{\theta,J}
 +\Delta^{(1)}_{\theta,n-J}+\bar D_{\theta,n},               \label{eq:delta-rec-d}
\end{align}
where $J\sim\mathrm{Bin}(n,1/2)$ and, conditional on the labelled first-bit
allocation, the two child samples are independent canonical samples.  The
root defect is a function of both children and is not independent of them.
Consequently, if $d(n)=\mathbb E\bar D_{\theta,n}$,
$a(n)=\mathbb E\Delta_{\theta,n}$, $q(n)=\operatorname{Var}\Delta_{\theta,n}$,
and $S_n=\Delta^{(0)}_{\theta,J}+\Delta^{(1)}_{\theta,n-J}$, then the exact
mean and variance recurrences are
\begin{align}
 a(n)&=2\mathbb Ea(B_n)+d(n),                                  \label{eq:amean-exact-d}\\
 q(n)&=2\mathbb Eq(B_n)+r(n),                                  \label{eq:qrec-d}\\
 r(n)&=\operatorname{Var}\{a(J)+a(n-J)\}
       +\operatorname{Var}(\bar D_{\theta,n})
       +2\operatorname{Cov}(S_n,\bar D_{\theta,n}).           \label{eq:signed-force-d}
\end{align}
Here $B_n\sim\mathrm{Bin}(n,1/2)$.  In particular the last covariance in
\eqref{eq:signed-force-d} may not be discarded or replaced by its absolute
value in an iteration.  \subsection{E.2--E.3: the two exact mean smoothings}
In this subsection put $d_n=\mathbb E\bar D_{\theta,n}$ (the bar is essential
at sizes zero and one) and define
\[
 ({\cal B}f)(n)=2^{-n}\sum_{k=0}^n{n\choose k}f(k),\qquad
 \rho_n=d_n-({\cal B}d)(n).
\]
This is an exact recurrence, $d_n=({\cal B}d)(n)+\rho_n$, not an
asymptotic one.  Couple the original sample and its rescaled middle-half
sample.  On $\mathcal G_n$ their barred defects agree (the guard implies
$n,K_n\ge2$), whereas off that event Cauchy--Schwarz, Proposition
\ref{prop:defect-moments-d}, and \eqref{eq:guardfail-d} give, for constants
$C<\infty,r<1$ depending only on $\theta$,
\begin{equation}|\rho_n|\le Cr^n.\label{eq:rho-d}\end{equation}
This argument does not condition the inward sample on the guard: its
unconditional law, conditional only on $K_n=k$, is canonical.

We give all smoothing details needed below.  Couple
$B_{n+1}=B_n+\xi$, where $B_n\sim\operatorname{Bin}(n,1/2)$ and $\xi$ is an
independent fair bit.  With $e_n=d_{n+1}-d_n$, subtraction gives
\begin{equation}e_n={1\over2}\mathbb Ee_{B_n}+\rho_{n+1}-\rho_n.\label{eq:e-d}\end{equation}
For $0<\beta<1$,
\begin{equation}(n+1)^\beta\mathbb E(B_n+1)^{-\beta}\longrightarrow2^\beta.
\label{eq:negmoment-d}\end{equation}
Indeed, on $|B_n/n-1/2|\le\epsilon$ the ratio is bounded and converges in
probability, and on the complement it is at most $(n+1)^\beta$ while
Chernoff bounds its probability by $2e^{-c_\epsilon n}$.  Thus the weighted
coefficient in \eqref{eq:e-d} tends to $2^{\beta-1}<1$.  Choose a strict
margin, isolate the self atom $B_n=n$ of coefficient $2^{-n-1}$, and choose
$C_\beta$ to dominate the finite initial range and the exponential forcing
divided by that margin.  Strong induction proves
\begin{equation}|d_{n+1}-d_n|\le C_{\theta,\beta}(n+1)^{-\beta}.\label{eq:dreg-d}\end{equation}
These choices specify the constants and range: first choose
$\gamma\in(2^{\beta-1},1)$, then $N$ so \eqref{eq:negmoment-d} and
$2^{-N-1}$ leave margin $1-\gamma$, and finally take the maximum of the
finitely many required ratios for $n<N$ and $2C(1+r)r^n(n+1)^\beta/(1-\gamma)$.

Taking expectations in \eqref{eq:delta-rec-d} gives exactly
$a(n)=2({\cal B}a)(n)+d_n$.  Put $z_n=a(n+1)-a(n)$.  The same coupling gives
\begin{equation}z_n=\mathbb Ez_{B_n}+e_n.\label{eq:z-d}\end{equation}
Iterate along half-thinnings $Z_{t+1}|Z_t\sim\operatorname{Bin}(Z_t,1/2)$
until $Z_t\le1$.  Uniform accumulation of the errors follows as follows.
For $t\le\lfloor\log_2n\rfloor-2$, Chernoff gives
\[
 \mathbb E[(Z_t+1)^{-\beta};Z_t>1]
 \le2^\beta(n2^{-t})^{-\beta}+e^{-n2^{-t}/8};
\]
for later $t$, $\mathbb P(Z_t>1)\le n^22^{-2t}/2$.  Both geometric sums are
uniformly bounded.  Hence $\sup_n|z_n|<\infty$.  Subtracting \eqref{eq:z-d}
once more, for $h_n=z_{n+1}-z_n$,
$h_n=\frac12\mathbb Eh_{B_n}+e_{n+1}-e_n$.  Estimate \eqref{eq:dreg-d} and
the same weighted induction prove $|h_n|\le C(n+1)^{-\beta}$, and telescoping
proves
\begin{equation}|z_u-z_v|\le C|u-v|(1+\min(u,v))^{-\beta}.\label{eq:zreg-d}\end{equation}
No decay of $z_n$ itself is asserted.

For $f_n(k)=a(k)+a(n-k)$, the midpoint is
$c_n=a(\lfloor n/2\rfloor)+a(\lceil n/2\rceil)$.  If $n=2m$,
\[
f_n(m+s)-c_n=\sum_{i=1}^s[z_{m+i-1}-z_{m-i}],
\]
and if $n=2m+1$ then $f_n(m)=f_n(m+1)=c_n$ and
\[
f_n(m+1+s)-c_n=\sum_{i=1}^s[z_{m+i}-z_{m-i}].
\]
Thus first-order terms cancel pairwise, including odd $n$.  On
$|k-n/2|\le n/4$, \eqref{eq:zreg-d} yields
\begin{equation}|f_n(k)-c_n|\le C(1+|k-n/2|)^2n^{-\beta}.\label{eq:pair-d}\end{equation}
Off this window bounded $z$ gives $Cn$, while Chernoff is exponential.
Since $\mathbb E|B_n-n/2|^4=3n^2/16-n/8$,
\begin{equation}\operatorname{Var}f_n(B_n)=O_{\theta,\eta}(n^\eta)\quad
\hbox{for every }\eta>0,\label{eq:meanvar-d}\end{equation}
by taking $\beta>1-\eta/2$.

\subsection{E.4--E.7: signed forcing and bootstrap}
Let $\mu_n(J)=f_n(J)$ and $b_n(J)=\mathbb E(\bar D_{\theta,n}|J)$.
Conditionally on $J$, the children are independent, so
$\mathbb E(S_n|J)=\mu_n(J)$ and
$\operatorname{Var}(S_n|J)=q(J)+q(n-J)$.  Total variance and total covariance
therefore expand the forcing, without an independence assumption, as
\begin{align}
r(n)={}&\operatorname{Var}(\mu_n(J))
 +\mathbb E\operatorname{Var}(\bar D_{\theta,n}|J)+\operatorname{Var}(b_n(J))\notag\\
&+2\mathbb E\operatorname{Cov}(S_n,\bar D_{\theta,n}|J)
 +2\operatorname{Cov}(\mu_n(J),b_n(J)).\label{eq:five-d}
\end{align}
This is exactly \eqref{eq:signed-force-d}.  With
$C_D=\sup_n\operatorname{Var}\bar D_{\theta,n}<\infty$, Cauchy--Schwarz and
\eqref{eq:meanvar-d} give
\begin{equation}|r(n)|\le Cn^\eta+C_D+2\sqrt{C_D(2\mathbb Eq(B_n)+Cn^\eta)}.
\label{eq:rboot-d}\end{equation}

For clarity we spell out the preliminary induction.  Fix $p=1+\epsilon$ and
choose $2^{1-p}<\gamma<1$.  The central-event proof used above also gives
$2\mathbb E(B_n+1)^p/(n+1)^p\to2^{1-p}$, so it is at most $\gamma$ beyond
some $n_0$.  Under $q(k)\le A(k+1)^p$ for $k<n$, isolate the self atom, whose
coefficient is $s_n=2^{1-n}$, to obtain
\[
\operatorname{Var}S_n\le s_nq(n)+\gamma A(n+1)^p+C_0(n+1)^\eta.
\]
In $q(n)\le V+C_D+2\sqrt{C_DV}$ use
$2\sqrt{C_Ds_nq(n)}\le\delta q(n)+C_Ds_n/\delta$, with
$\delta<(1-\gamma)/4$, enlarge $n_0$ until $s_n<(1-\gamma)/4$, and take
$\eta<p$.  The lower powers are absorbed by the remaining strict margin;
then enlarge $A$ for $n<n_0$.  Strong induction proves
\begin{equation}q(n)=O_{\theta,\epsilon}(n^{1+\epsilon}).\label{eq:qcoarse-d}\end{equation}
There is no use here of a linear bound.  Substitution in \eqref{eq:rboot-d},
choosing the coarse and mean exponents sufficiently small, yields
\begin{equation}|r(n)|=O_{\theta,\epsilon}(n^{1/2+\epsilon})
\quad(\epsilon>0).\label{eq:rsub-d}\end{equation}

\subsection{E.8: signed binomial renewal}
\begin{theorem}[signed renewal]\label{thm:renew-d}
If $q(0)=q(1)=r(0)=r(1)=0$, $q(n)=2\mathbb Eq(B_n)+r(n)$, and
$|r(n)|\le Cn^\alpha$ for $n\ge2$ with $0<\alpha<1$, then $q(n)=O(n)$.
\end{theorem}
\begin{proof}
Repeated independent thinning gives
\[
q(n)=\sum_{j<L}2^j\mathbb Er(\operatorname{Bin}(n,2^{-j}))
 +2^L\mathbb Eq(\operatorname{Bin}(n,2^{-L})).
\]
The boundary tends to zero: for fixed $n$, the last expectation is
$O_n(2^{-2L})$ because $q(0)=q(1)=0$ and
$\mathbb P\{\operatorname{Bin}(n,p)\ge2\}\le n(n-1)p^2/2$.  Hence the
infinite forcing representation exists.  It is absolutely convergent and
$O(n)$: put $\lambda=n2^{-j}$.  For $\lambda\ge1$, concavity gives
$\mathbb E B^\alpha\le\lambda^\alpha$, so
$2^j\mathbb E|r(B)|\le Cn\lambda^{\alpha-1}$; summing over the dyadic
$\lambda\ge1$ is $O(n)$.  For $\lambda<1$, since $r(0)=r(1)=0$ and
$k^\alpha\le k(k-1)$ for $k\ge2$,
$2^j\mathbb E|r(B)|\le C2^jn(n-1)2^{-2j}\le Cn\lambda$, whose tail is also
$O(n)$.  These absolute bounds justify every signed cancellation.  They
also control all boundary terms.  In particular, any dyadic log-periodic
component is bounded after division by $n$ by this same absolute majorant;
no monotonicity, Poissonization, or hidden periodic assumption is used.
\end{proof}
Choose $\alpha\in(1/2,1)$ in \eqref{eq:rsub-d}.  The theorem proves
\begin{equation}\boxed{\operatorname{Var}\Delta_{\theta,n}=O_\theta(n).}
\label{eq:renew-result-d}\end{equation}

The following stabilization proof is retained as an independent check, but
is not used in E.2--E.8.

For a sample $Y=(Y_1,\ldots,Y_k)$ and an independent uniform point $X$, put
\[
 Z_k=D_\theta(Y_1,\ldots,Y_k,X)-D_\theta(Y_1,\ldots,Y_k),
\]
with the barred convention at sizes zero and one.  Write $I^*=[1/4,3/4)$,
let $I^{*j}$ be its $j$-fold affine inward copy, let
$S_j=\{X\in I^{*j}\}$, and let $G_j$ be the event that the old points in
$I^{*j}$ occupy every one of the $M$ guard subcells after affine rescaling.
Thus $S_0$ is certain, $|I^{*j}|=2^{-j}$, and $G_j$ depends only on the old
sample.

\begin{lemma}[support of a root add-one cost]\label{lem:uniform-support-d}
For every integer $L\ge0$,
\begin{equation}
 \{Z_k\ne0\}\subset S_L\cup\bigcup_{0\le j<L}(S_j\cap G_j^c). \label{eq:uniform-support-d}
\end{equation}
Moreover, for constants depending only on $\theta$,
\begin{equation}
 \mathbb P(Z_k\ne0)\le {C_\theta\over k+1}.                   \label{eq:add-support-prob-d}
\end{equation}
\end{lemma}
\begin{proof}
For $L=0$, \eqref{eq:uniform-support-d} says only that the event is contained
in the whole sample space.  Suppose it holds at $L$.  On
$S_L\cap G_L$, apply Theorem~\ref{thm:inward-copy-dossier} once to the old
configuration and once after insertion.  If $X\notin I^{*(L+1)}$, the two
inward configurations are identical, so their defects, and hence their
difference, are identical.  If $X\in I^{*(L+1)}$, no conclusion is needed.
On $S_L\cap G_L^c$ we retain the guard-failure event.  This replaces the term
$S_L$ by $(S_L\cap G_L^c)\cup S_{L+1}$ and proves the induction step.  Notice
that this compares the old and new configurations under the old guard; adding
a point cannot destroy that guard.

Let $N_j$ be the old occupancy of $I^{*j}$.  Conditional on $N_j=s$, the
$s$ old points there are iid uniform, so a union bound over the $M$ guard
subcells gives
\[
 \mathbb P(G_j^c\mid N_j=s)\le M(1-1/M)^s.
\]
Since $N_j\sim\mathrm{Bin}(k,2^{-j})$, the binomial generating function gives
\[
 \mathbb P(G_j^c)\le M(1-2^{-j}/M)^k\le M e^{-k2^{-j}/M}.
\]
The new point is independent of the old sample, hence $S_j$ and $G_j$ are
independent.  Letting $L\to\infty$ in \eqref{eq:uniform-support-d} (the
nested intervals have intersection of Lebesgue measure zero) yields
\[
 \mathbb P(Z_k\ne0)\le M\sum_{j\ge0}2^{-j}e^{-k2^{-j}/M}.
\]
For $k\ge1$, put $j_0=\lfloor\log_2 k\rfloor$.  For $j>j_0$ the sum of
$2^{-j}$ is at most $2/k$.  For $j\le j_0$, set $l=j_0-j$ and use
$k2^{-j}\ge2^l$ and $2^{-j}\le2^{l+1}/k$; thus this part is at most
$2k^{-1}\sum_{l\ge0}2^le^{-2^l/M}$.  The latter series converges.  Enlarging
the constant handles $k=0$ and proves \eqref{eq:add-support-prob-d}.
\end{proof}

\begin{lemma}[uniform local stabilization]\label{lem:uniform-localstab-d}
For every $0<\alpha<1$ there is $C_{\theta,\alpha}<\infty$ such that
\begin{equation}
 \mathbb E|Z_k|^2\le C_{\theta,\alpha}(k+1)^{-\alpha}.         \label{eq:uniform-localstab-d}
\end{equation}
\end{lemma}
\begin{proof}
Choose $p>2/(1-\alpha)$.  Proposition~\ref{prop:defect-moments-d}, applied
at sizes $k$ and $k+1$, and $|u-v|^p\le2^{p-1}(|u|^p+|v|^p)$ give
$\sup_k\mathbb E|Z_k|^p<\infty$.  By H\"older and
\eqref{eq:add-support-prob-d},
\[
 \mathbb E|Z_k|^2
 \le(\mathbb E|Z_k|^p)^{2/p}\mathbb P(Z_k\ne0)^{1-2/p}
 \le C(k+1)^{-(1-2/p)}\le C(k+1)^{-\alpha}.
\]
\end{proof}

\begin{lemma}[bounded discrepancy add-one cost]\label{lem:delta-addone-d}
If $X_1,\ldots,X_m,X$ are iid uniform and
\[
 A_m^+=\Delta_\theta(X_1,\ldots,X_m,X)
       -\Delta_\theta(X_1,\ldots,X_m),
\]
then $\sup_m\mathbb E|A_m^+|^2<\infty$.
\end{lemma}
\begin{proof}
For the depth-$r$ cell $I_r(X)$ let $K_r$ be its old occupancy.  Subtracting
the two finite-tree sums \eqref{eq:finite-defect-sum} cancels every node not
on the ancestor chain of $X$ and gives, pathwise,
\begin{equation}
 A_m^+=\sum_{r\ge0}\delta_r,\qquad
 \delta_r=\bar D_\theta(S_{I_r(X)}\cup\{X\})
             -\bar D_\theta(S_{I_r(X)}).                     \label{eq:add-chain-d}
\end{equation}
After affine rescaling, conditional on the identity of $I_r(X)$ and on
$K_r=k$, the old points in that cell and $X$ are independent iid uniform
points in the cell: this follows directly by conditioning their independent
binary prefixes and then retaining their independent suffixes.  Hence Lemma
\ref{lem:uniform-localstab-d} gives
\begin{equation}
 \mathbb E(\delta_r^2\mid K_r)\le C(K_r+1)^{-\alpha}.        \label{eq:delta-large-d}
\end{equation}
Here $K_r\sim\mathrm{Bin}(m,2^{-r})$ and $\lambda_r=m2^{-r}$.

We record both elementary binomial estimates used below.  If $K\sim
\mathrm{Bin}(m,p)$ and $\lambda=mp\ge1$, Chernoff's bound gives
$\mathbb P(K<\lambda/2)\le e^{-\lambda/8}$, whence
\[
 \mathbb E(K+1)^{-\alpha}
 \le(\lambda/2)^{-\alpha}+e^{-\lambda/8}\le C_\alpha\lambda^{-\alpha}.
\]
(The displayed Chernoff inequality follows from
$\mathbb Ee^{-tK}=(1-p+pe^{-t})^m\le
\exp[-\lambda(1-e^{-t})]$ and Markov's inequality, taking $t=\log2$ and
weakening the resulting absolute constant.)  Thus, when $\lambda_r\ge1$,
\begin{equation}\|\delta_r\|_2\le C\lambda_r^{-\alpha/2}.     \label{eq:large-lambda-d}\end{equation}
When $\lambda_r\le1$, $K_r=0$ implies $\delta_r=0$, because both barred
root defects then vanish.  A uniform fourth moment of $\delta_r$ follows
from Proposition~\ref{prop:defect-moments-d}; Cauchy--Schwarz and
$\mathbb P(K_r\ge1)\le\mathbb EK_r=\lambda_r$ give
\begin{equation}\|\delta_r\|_2\le C\lambda_r^{1/4}.           \label{eq:small-lambda-d}\end{equation}
Minkowski's inequality, \eqref{eq:large-lambda-d}, and
\eqref{eq:small-lambda-d} now yield
\[
 \|A_m^+\|_2\le\sum_{r\ge0}\|\delta_r\|_2
 \le C\sum_{r:\lambda_r\ge1}\lambda_r^{-\alpha/2}
      +C\sum_{r:\lambda_r<1}\lambda_r^{1/4}\le C_{\theta,\alpha}.
\]
Both sums are geometric when indexed away from the unique scale at which
$\lambda_r$ crosses one.  Formula \eqref{eq:add-chain-d} is initially a
finite sum up to the isolation depth of $X$; the displayed summable bound
also proves convergence in $L^2$, so no interchange of an undominated
infinite sum is being made.
\end{proof}

\begin{theorem}[linear adaptive discrepancy variance]\label{thm:delta-linear-d}
For fixed $\theta$, $\operatorname{Var}\Delta_{\theta,n}=O_\theta(n)$.
\end{theorem}
\begin{proof}
Let $X^{(i)}$ be obtained by replacing $X_i$ by an independent copy $X_i'$.
Relative to the common sample with label $i$ removed, the difference
$\Delta(X)-\Delta(X^{(i)})$ is one insertion cost minus another.  Lemma
\ref{lem:delta-addone-d} and $(u-v)^2\le2u^2+2v^2$ therefore give
\[
 \sup_{n,i}\mathbb E\{\Delta(X)-\Delta(X^{(i)})\}^2\le C_\theta.
\]
The Efron--Stein inequality, in its replacement form,
$\operatorname{Var}W\le\frac12\sum_i\mathbb E(W-W^{(i)})^2$, now gives
$\operatorname{Var}\Delta_{\theta,n}\le C_\theta n$.
This proof also controls all covariance terms in \eqref{eq:signed-force-d}
without assuming independence and makes a signed renewal argument unnecessary.
\end{proof}

\section{Uniform nonresonant full theorem}
Assume $c(\theta)>0$.  The certified strict-stopped result above gives
\[
 \operatorname{Var}V_{\theta,n}\sim c(\theta)n\log_2n,
 \qquad
 {V_{\theta,n}-\mathbb EV_{\theta,n}\over
  \sqrt{c(\theta)n\log_2n}}\Rightarrow N(0,1).
\]
The recurrence module above supplies
$\operatorname{Var}\Delta_{\theta,n}=O_\theta(n)$.  No independence
between $V_{\theta,n}$ and $\Delta_{\theta,n}$ is needed.  Indeed,
Cauchy--Schwarz gives, explicitly,
\[
 |\operatorname{Cov}(V_{\theta,n},\Delta_{\theta,n})|
 \le \sqrt{\operatorname{Var}V_{\theta,n}
                  \operatorname{Var}\Delta_{\theta,n}}
 =O_\theta(n\sqrt{\log n})=o(n\log n).
\]
Since $T_{\theta,n}=V_{\theta,n}+\Delta_{\theta,n}$, expansion of its
variance therefore gives
\[
 \operatorname{Var}T_{\theta,n}
 =\operatorname{Var}V_{\theta,n}+\operatorname{Var}\Delta_{\theta,n}
   +2\operatorname{Cov}(V_{\theta,n},\Delta_{\theta,n})
 \sim c(\theta)n\log_2n.
\]
Moreover
\[
 {\Delta_{\theta,n}-\mathbb E\Delta_{\theta,n}\over
       \sqrt{c(\theta)n\log_2n}}\longrightarrow0
 \quad\hbox{in }L^2,
\]
so a single application of Slutsky's theorem gives the centered CLT
with deterministic normalization; the displayed variance equivalence and
another application of Slutsky gives true-variance normalization.
There is no interchange of limits in this transfer.

The complete recurrence, midpoint pairing, noncircular bootstrap, and
signed-renewal proof are E.2--E.8 above.  Together with the independently
certified strict-stopped theorem, the displayed Cauchy--Schwarz estimate and
the two applications of Slutsky certify the uniform nonresonant full theorem,
including true-variance normalization.

\section{Resonant stopped and adaptive theory: dependency-complete statement}
\label{sec:resonant-dossier}
Fix $Q\ge3$ and $\theta=2/Q$, and retain the notation of
Sections~1--2.  Thus $b=(Q-1)/2$, $J=\lceil b\rceil$, and
\[
 r_j=\max\{\lceil\log _2(j+1)\rceil,
                 \lceil\log _2(b-j)\rceil\},\qquad
 c_j=(b-j)2^{-r_j}.
\]
The transfer used everywhere in this section is not an unspecified
cohomologous representative.  Put $D(c)=\{2c\}$, $S_c=\bar R_c$ for $0<c\le1$, extend
$S_0:=0$, and
\begin{align}
 \bar A_Q&=A_Q-\mathbb EA_Q
   =\sum_{j<J}\sum_{\ell\ge0}S_{c_j2^{-\ell}},\label{eq:Abar-explicit-d}\\
 K_Q&=\sum_{j<J}\sum_{s\ge1}2^{-s}S_{D^s(c_j)},\qquad
 g_Q=-\bar A_Q-K_Q.                                   \label{eq:g-explicit-d}
\end{align}
The first equality in \eqref{eq:Abar-explicit-d} is in every finite $L^p$;
it is the centered version of the $A$ in Section~2.  The $K_Q$ here is
exactly the function denoted $k$ there, specialized to $\theta=2/Q$.
Indeed $Q=2b+1=\mathbb EF_{2/Q}$, and the Fourier criterion in Section~3
says that $d=0$ at this parameter.  Equations (2.4)--(2.5) consequently give
$F_{2/Q}-Q=h\circ T-h$ with $h=\bar A_Q+K_Q$, proving, rather than merely
choosing, that $g_Q=-h$ and
\begin{equation}F_{2/Q}-Q=g_Q-g_Q\circ T,\qquad \mathbb Eg_Q=0,\quad g_Q\in L^p\ (p<\infty). \label{eq:res-cob}
\end{equation}

\begin{lemma}[Explicit resonant Perron orbit and parity translation]
\label{lem:explicit-perron-parity-d}
Let $R_Q=\max_{j<J}r_j$.  Then, as $L^p$ identities for every finite $p$,
\begin{align}
 P\bar A_Q&={1\over2}\bar A_Q+{1\over2}\sum_{j<J}S_{D(c_j)},
                                                        \label{eq:P-Abar-d}\\
 PK_Q&=\sum_{j<J}\sum_{s\ge2}2^{-s}S_{D^s(c_j)},       \label{eq:P-K-d}\\
 Pg_Q&={1\over2}g_Q-{1\over2}K_Q,\qquad
 P^{R_Q+1}K_Q=0.                                      \label{eq:Pg-exact-d}
\end{align}
Consequently, if $Q$ is odd and
$W_Q:=Pg_Q-g_Q/2$, then
\[
 W_Q=-K_Q/2,\qquad P^{R_Q+1}W_Q=0.
\]
If $Q$ is even and $K_Q:=g_Q-2Pg_Q$ (the notation already used in the mixed
recurrence), this is the very same explicit $K_Q$ in
\eqref{eq:g-explicit-d}, and $P^{R_Q+1}K_Q=0$.  Thus one may take
$L_Q=R_Q+1$ in both parities (the minimal depth may be smaller through
cancellation).
\end{lemma}
\begin{proof}
Direct averaging of the two inverse branches, including the overlap when
$c>1/2$, gives
\begin{equation}
 PS_c={1\over2}S_{D(c)}.                              \label{eq:PS-D-d}
\end{equation}
For example, if $c<1/2$ this is $S_{2c}/2$; if $c>1/2$ the raw answer is
$-S_{2-2c}/2=S_{2c-1}/2$; and at $c=1/2$ both sides are zero a.e.  This
case division derives \eqref{eq:PS-D-d} from the actual step functions and
also shows that only their finitely many endpoints are discarded.

Apply \eqref{eq:PS-D-d} term by term first to finite partial sums of
\eqref{eq:Abar-explicit-d}.  For $\ell\ge1$,
$D(c_j2^{-\ell})=c_j2^{-(\ell-1)}$; separating the $\ell=0$ term and then
passing in $L^p$ gives \eqref{eq:P-Abar-d}; this passage is valid because
$P$ is a contraction on $L^p$.  Applying the same formula to
the uniformly convergent series defining $K_Q$ and shifting $s$ gives
\eqref{eq:P-K-d}.  Since
\[
 K_Q={1\over2}\sum_{j<J}S_{D(c_j)}
       +\sum_{j<J}\sum_{s\ge2}2^{-s}S_{D^s(c_j)},
\]
substitution into $g_Q=-\bar A_Q-K_Q$ proves
$Pg_Q=g_Q/2-K_Q/2$.  Iteration is therefore completely explicit:
\begin{equation}
 P^ag_Q=2^{-a}g_Q-\sum_{r=0}^{a-1}2^{-(a-r)}P^rK_Q,qquad
 P^aK_Q=2^{-a}\sum_{j<J}\sum_{s\ge1}2^{-s}S_{D^{s+a}(c_j)}.
                                                        \label{eq:P-iterates-d}
\end{equation}
Now
$c_j=(Q-1-2j)/2^{r_j+1}$.  Hence $D^{r_j+1}(c_j)=0$ on the circle,
$S_0=0$, and every summand in the last display vanishes once
$a\ge R_Q+1$.  This is the advertised cancellation; the depth depends only
on $Q$, never on the sample size.  Finally the definitions of $W_Q$ and of
the even-parity $K_Q$ applied to \eqref{eq:Pg-exact-d} give the two stated
identities and signs.
\end{proof}
The odd/even difference is therefore one of coordinates, not of the
underlying transfer: odd root-toll notation stores the boundary correction
as $W_Q=Pg_Q-g_Q/2=-K_Q/2$, whereas even notation stores twice its negative
as $K_Q=g_Q-2Pg_Q$.  Formula \eqref{eq:P-iterates-d} is an explicit finite
dyadic-step orbit, so no spectral or ``finite Walsh space'' assertion is
being used in its place.

For the strict isolation depth, the pathwise stopped identity is
\begin{equation}V_{2/Q,n}=QL_n+\sum_{i=1}^n\{g_Q(X_i)-g_Q(T^{\tau_i}X_i)\}. \label{eq:res-stop-id}
\end{equation}
The following fills the first PR~61 gap; in particular, it does not use the
false shortcut ``prefix-free, therefore independent.''

\begin{lemma}[Labelled stopped-line cylinder factorization]
\label{lem:res-suffix-factor}
Write $X_i=(X_i(1),X_i(2),\ldots)\in\{0,1\}^{\mathbb N}$, choosing either
binary expansion on the null set of dyadic rationals, and put
\[
 \tau_i=\min\{r\ge0:X_i(1)\cdots X_i(r)\hbox{ is shared by no }j\ne i\},
 \quad W_i=X_i(1)\cdots X_i(\tau_i),\quad Y_i=T^{\tau_i}X_i.
\]
On the null event that two infinite strings coincide define these objects
arbitrarily.  Let
\[
 \mathcal G_n=\sigma(W_1,\ldots,W_n)
 =\sigma\{\tau_i,X_i(1),\ldots,X_i(\tau_i):i\le n\}.
\]
Conditionally on $\mathcal G_n$, the vector $(Y_1,\ldots,Y_n)$ has product
fair-bit law, and hence is independent of $\mathcal G_n$.  The same statement
holds conditionally on
\[
 \mathcal P_\ell=\sigma\{\tau_i\wedge\ell,
 X_i(1),\ldots,X_i(\tau_i\wedge\ell),
 {\bf1}_{\{\tau_i\le\ell\}}:i\le n\}.
\]
\end{lemma}
\begin{proof}
Call a labelled tuple $w=(w_1,\ldots,w_n)$ admissible when (i) its words are
prefix-free and (ii), for every $i$ and $0\le r<|w_i|$, some $j\ne i$ has
$|w_j|>r$ and $w_j|_r=w_i|_r$.  Condition (i) says that label $i$ is alone
at depth $|w_i|$; condition (ii) says it was not alone at any earlier depth.
Thus, writing $E_w=\{(W_i)_i=w\}$, for every admissible $w$ one has the exact
atom identity
\begin{equation}
 E_w=\bigcap_{i=1}^n\{X_i(1)\cdots X_i(|w_i|)=w_i\}.       \label{eq:stop-line-atom}
\end{equation}
For a nonadmissible tuple $E_w$ is empty.  For $n=1$ the unique admissible
tuple consists of the empty word; for $n\ge2$ no terminal word is empty.

Let $i_1,\ldots,i_k$ be distinct labels and let $C_a=[u_a]$ be arbitrary
finite suffix cylinders.  All coordinates fixed in
\eqref{eq:stop-line-atom} and all coordinates fixed by the $C_a$ are
distinct.  Product measure therefore gives, exactly,
\begin{align*}
 &\mathbb P\{E_w,\ Y_{i_a}\in C_a\ (1\le a\le k)\}\\
 &\qquad=2^{-\sum_i|w_i|-\sum_a|u_a|}
 =\mathbb P(E_w)\prod_{a=1}^k2^{-|u_a|}.
\end{align*}
Equivalently, on every positive-probability stopped-line atom,
\[
 \mathbb P\{Y_{i_1}\in C_1,\ldots,Y_{i_k}\in C_k\mid\mathcal G_n\}
 =\prod_{a=1}^k2^{-|u_a|}.
\]
This calculation explains why the random depths cause no stopping-time
difficulty: on $E_w$ the depths are the fixed integers $|w_i|$, and the atom
inspects no later coordinate.  (Repeated labels are handled by first
intersecting their cylinder requirements; the displayed product assertion is
of course for distinct coordinates.)

There are countably many finite labelled tuples, and their admissible atoms
partition the probability-one event of pairwise distinct input strings.
Summing over atoms proves the factorization with any event in $\mathcal G_n$.
A $\pi$--$\lambda$ argument, first in each suffix coordinate and then in
their finite product, extends it from cylinders to the full product suffix
sigma-field.  Finally $\mathcal P_\ell\subseteq\mathcal G_n$; taking conditional
expectations down from $\mathcal G_n$ proves the assertion for
$\mathcal P_\ell$.  Conditioning only on an unlabelled terminal family would
not justify this last conclusion, which is why both fields retain labels.
\end{proof}

\subsection{The stopped-trie variance phase: a direct second-moment renewal}
\label{subsec:stopped-trie-phase}

We now supply the fixed-size phase theorem used by the stopped-vector
argument.  This is a variance calculation, not a consequence of the CLT.
The sample is the canonical one of independent fair infinite binary strings
$X_1,\ldots,X_n$.  With $\tau_i$ as in
Lemma~\ref{lem:res-suffix-factor}, put
\begin{equation}
 L_n=\sum_{i=1}^n\tau_i,\qquad L_0=L_1=0.       \label{eq:trie-L-definition}
\end{equation}
Thus $L_n$ is the \emph{external} path length of the strict stopped binary
trie: a label contributes one for every edge traversed before its first
isolating prefix.  It is not the internal path length.  In particular the
unique label in a one-point sample is isolated at the empty prefix and has
$\tau_1=0$.

\begin{theorem}[Stopped-trie phase variance and uniform tail]
\label{thm:stopped-trie-phase}
Let $m(n)=\mathbb EL_n$ and $v(n)=\operatorname {Var}L_n$.  There is an
explicit bounded continuous one-periodic function $v_L$ such that
\begin{equation}
 \frac{v(n)}n\longrightarrow v_L(t)
 \quad\hbox{whenever }\{\log _2n\}\longrightarrow t.       \label{eq:trie-phase-limit}
\end{equation}
Moreover,
\begin{equation}
 \sup_{k\ge M}\left|\frac{v(k)}k-v_L(\{\log _2k\})\right|
 \longrightarrow0\qquad(M\to\infty).                       \label{eq:trie-uniform-tail}
\end{equation}
Both assertions follow directly from the exact variance recurrence below.
\end{theorem}

\begin{proof}
\emph{Exact split and conditional-mean forcing.}
For $n\ge2$, let $J_n$ be the number of first bits equal to zero.  Thus
$J_n\sim\operatorname {Bin}(n,1/2)$, and, after removal of the first bit,
the two labelled child samples are conditionally independent canonical
samples.  Every label is nonisolated at the root and traverses exactly one
first edge.  Consequently, including the self-split events $J_n=0,n$,
\begin{equation}
 L_n\ \mathrel{\mathop=\limits^{d}}\
 L_{J_n}^{(0)}+L_{n-J_n}^{(1)}+n.             \label{eq:trie-distributional-rec}
\end{equation}
The toll is exactly $n$.  Taking expectations gives
\begin{equation}
 m(n)=2\mathbb Em(B_n)+n\quad(n\ge2),\qquad m(0)=m(1)=0,     \label{eq:trie-mean-rec}
\end{equation}
where $B_n\sim\operatorname {Bin}(n,1/2)$.  If
$X_n=L_n-m(n)$, then the exact centered recurrence is
\begin{equation}
 X_n=X_{J_n}^{(0)}+X_{n-J_n}^{(1)}+d_n(J_n),\qquad
 d_n(j)=m(j)+m(n-j)+n-m(n),                  \label{eq:trie-centered-rec}
\end{equation}
and $\mathbb Ed_n(J_n)=0$.  Conditional variance in
\eqref{eq:trie-distributional-rec} therefore gives
\begin{align}
 v(n)&=2\mathbb Ev(B_n)+r_L(n),                         \label{eq:trie-variance-rec}\\
 r_L(n)&=\operatorname {Var}\{m(B_n)+m(n-B_n)\},        \label{eq:trie-force}
\end{align}
with $v(0)=v(1)=r_L(0)=r_L(1)=0$.  Formula
\eqref{eq:trie-force} is precisely the variance of the conditional mean;
the deterministic toll cancels, but this term does not.

\emph{Mean formula and regularity.}
A fixed label has $\tau_i>r$ precisely when at least one of the other labels
shares its prefix of length $r$.  Survival counting and Tonelli's theorem
give, for $n\ge2$,
\begin{equation}
 m(n)=n\sum_{r\ge0}\{1-(1-2^{-r})^{n-1}\}.              \label{eq:trie-mean-formula}
\end{equation}
Put $p_r=2^{-r}$ and
\[
 S_a=\sum_{r\ge0}\{1-(1-p_r)^a\},\quad
 A_a=\sum_{r\ge0}p_r(1-p_r)^a,\quad
 D_a=\sum_{r\ge0}p_r^2(1-p_r)^a.
\]
For $e(n)=m(n+1)-m(n)$, direct subtraction of
\eqref{eq:trie-mean-formula} gives, for $n\ge1$,
\begin{equation}
 e(n)=S_n+nA_{n-1},\qquad
 e(n+1)-e(n)=2A_{n-1}-(n+2)D_{n-1}.          \label{eq:trie-mean-differences}
\end{equation}
For $a=0$ the two sums below are finite geometric sums.  For $a\ge1$,
$(1-p)^a\le e^{-ap}\le e^{-(a+1)p/2}$; splitting the dyadic values
$y=(a+1)2^{-r}$ at $y=1$ yield
\[
 A_a\le {1\over a+1}\sum_{r\ge0}y e^{-y/2}\le {C\over a+1},
 \qquad
 D_a\le {1\over(a+1)^2}\sum_{r\ge0}y^2e^{-y/2}\le {C\over(a+1)^2}.
\]
Indeed below one each sum is geometric, while above one its dyadic
exponential tail is summable, uniformly in $a$.  Hence
\begin{equation}
 |e(n+1)-e(n)|\le {C\over n+1}.                         \label{eq:trie-second-difference}
\end{equation}

We use this estimate without hiding the midpoint parity.  Set
$f_n(j)=m(j)+m(n-j)$.  On $n/4\le j\le3n/4$,
\[
 f_n(j+1)-f_n(j)=e(j)-e(n-j-1),qquad
 |f_n(j+1)-f_n(j)|\le C{|2j+1-n|+1\over n}.
\]
The first difference is zero at the symmetric odd midpoint; for even $n$
the displayed extra one covers the two adjacent indices.  Summation from
$\lfloor n/2\rfloor$ in either parity gives
\begin{equation}
 |f_n(j)-f_n(\lfloor n/2\rfloor)|
 \le C\left(1+{|j-n/2|^2\over n}\right).                \label{eq:trie-midpoint-bound}
\end{equation}
The centered binomial fourth moment is $O(n^2)$, so the square of the
right side has bounded expectation on the central window.  Also
\eqref{eq:trie-mean-formula} gives $m(k)\le Ck\log(k+1)$: terms with
$2^{-r}\ge1/k$ contribute $O(k\log(k+1))$, and the remaining geometric
tail contributes $O(k)$.  A binomial Chernoff bound therefore makes the
contribution of $B_n\notin[n/4,3n/4]$ exponentially small even after
squaring this polynomial-logarithmic bound.  Since variance is bounded by
the mean square about any constant, \eqref{eq:trie-midpoint-bound} proves
\begin{equation}0\le r_L(n)\le C\quad\hbox{for every }n.  \label{eq:trie-force-bounded}
\end{equation}

\emph{Exact thinning renewal and its boundary.}
Iterating \eqref{eq:trie-variance-rec} $q$ times, using binomial thinning,
gives the finite identity
\begin{equation}
 v(n)=\sum_{j=0}^{q-1}2^j\mathbb E r_L(\operatorname {Bin}(n,2^{-j}))
 +2^q\mathbb Ev(\operatorname {Bin}(n,2^{-q})).          \label{eq:trie-finite-renewal}
\end{equation}
For fixed $n$, let $C_n=\max_{0\le k\le n}v(k)<\infty$.  Because
$v(0)=v(1)=0$ and
$\mathbb P\{\operatorname {Bin}(n,p)\ge2\}\le {n\choose2}p^2$,
the last term is at most $C_n{n\choose2}2^{-q}$ and tends to zero.  Thus
the self-atoms in the split leave no boundary term, and
\begin{equation}
 v(n)=\sum_{j\ge0}2^j\mathbb E r_L(\operatorname {Bin}(n,2^{-j}))     \label{eq:trie-infinite-renewal}
\end{equation}
exactly.  The same estimate proves absolute convergence.

\emph{Poisson limiting forcing and the phase function.}
For $P_x\sim\operatorname {Pois}(x)$ define
\begin{equation}
 R_L(x)=\mathbb E r_L(P_x),\qquad
 v_L(t)=\sum_{a\in\mathbb Z}{R_L(2^{t-a})\over2^{t-a}}. \label{eq:trie-phase-series}
\end{equation}
The bounded sequence $r_L(k)$ makes the Poisson power series for $R_L$
locally uniformly convergent, hence $R_L$ is continuous.  Since $r_L(0)$
and $r_L(1)$ vanish,
\begin{equation}
 R_L(x)\le Cx^2\quad(0<x\le1),\qquad R_L(x)\le C\quad(x\ge1).          \label{eq:trie-poisson-bounds}
\end{equation}
Uniformly for $t\in[0,1]$, the summands in
\eqref{eq:trie-phase-series} are bounded by
$C\min(2^{t-a},2^{a-t})$, which has a bilateral geometric envelope.
The series therefore converges normally.  It defines a bounded continuous
function on a fundamental interval, and reindexing $a$ after replacing
$t$ by $t+1$ proves one-periodicity (and continuity at the identified
endpoints).

\emph{Fixed-size phase convergence.}
Let $q_n=\lfloor\log _2n\rfloor$, $t_n=\{\log _2n\}$, and
$x_{n,a}=n2^{-(q_n+a)}=2^{t_n-a}$.  Reindexing
\eqref{eq:trie-infinite-renewal} by $j=q_n+a$ gives
\begin{equation}
 {v(n)\over n}=\sum_{a\ge-q_n}{
 \mathbb E r_L(\operatorname {Bin}(n,x_{n,a}/n))\over x_{n,a}}.      \label{eq:trie-reindexed}
\end{equation}
For fixed $a$, if $t_n\to t$, binomial-to-Poisson convergence and the
boundedness of $r_L$ make the summand converge to the corresponding term of
\eqref{eq:trie-phase-series}.  Domination here is for the actual binomial
array, not merely its Poisson limit.  If $Z\sim\operatorname {Bin}(n,x/n)$,
then \eqref{eq:trie-force-bounded}, $r_L(0)=r_L(1)=0$, and
$\mathbb P(Z\ge2)\le x^2/2$ give
\begin{equation}
 {\mathbb E r_L(Z)\over x}\le C\min(x,x^{-1}).           \label{eq:trie-binomial-majorant}
\end{equation}
For $t_n\in[0,1)$ this has a fixed bilateral geometric envelope in $a$;
terms $a<-q_n$ are read as zero.  Dominated convergence in
\eqref{eq:trie-reindexed} proves \eqref{eq:trie-phase-limit}, independently
of every distributional limit theorem.

\emph{Uniform phase tail.}
If \eqref{eq:trie-uniform-tail} failed, there would be an $\epsilon>0$ and
integers $k_j\to\infty$ for which the displayed error is at least
$\epsilon$.  Compactness of the phase circle gives a subsequence whose
phases converge to some $t$ (with endpoints identified).  The already proved
phasewise limit sends $v(k_j)/k_j$ to $v_L(t)$, while continuity and
periodicity send $v_L(\{\log _2k_j\})$ to the same number.  This
contradiction proves the claimed supremum tail.
\end{proof}

The theorem has the normalization required below: its $v_L(t)$ is the limit
of $n^{-1}\operatorname {Var}L_n$, not a variance obtained from a CLT.  At a
half-log cut with $K=2^\ell$, $\lambda=n/K\to\infty$, and multinomial
occupancies $(N_I)$, a Chernoff union bound gives
$\max_I|N_I/\lambda-1|\to0$ in probability.  Thus the occupancies tend
uniformly to infinity and their phases approach that of
$\lambda$ (which equals that of $n$).  Equation
\eqref{eq:trie-uniform-tail}, uniform continuity of $v_L$, and
$\sum_I N_I=n$ give
\begin{equation}
 {1\over n}\sum_Iv(N_I)\longrightarrow v_L(t)
 \quad\hbox{in probability whenever }\{\log _2n\}\to t. \label{eq:trie-occupancy-transfer}
\end{equation}
Consequently the $(1,1)$ entry of the stopped vector's covariance is
$v_L(t)$; its other two diagonal entries remain the independently computed
$\sigma_Q^2$.  For the stopped-cost block matrix $C_Q(m)$ used in the
random-occupancy transfer, the same statement supplies exactly the
$Q^2v_L(t)+\sigma_Q^2$ part of its $(1,1)$ entry.  No assertion about its
mixed or discrepancy entries is made here.

\subsection{Ingredient 2: the joint stopped vector}
Put $\sigma_Q^2=\mathbb E g_Q^2$ and, with the stopping depths and suffixes
of Lemma~\ref{lem:res-suffix-factor}, define
\[
 A_n=\sum_{i=1}^ng_Q(X_i),\qquad B_n=\sum_{i=1}^ng_Q(Y_i),
 \qquad Y_i=T^{\tau_i}X_i.
\]
There is no fourth coordinate.  The exact pathwise identity (also for
$n=0,1$) is
\begin{equation}
 V_{2/Q,n}=QL_n+A_n-B_n.                         \label{eq:stopped-vector-id}
\end{equation}
For $n=1$, $\tau_1=0$ and $A_1=B_1$, so this convention has no hidden
singleton correction.

\begin{theorem}[Joint stopped-vector Cram\'er--Wold theorem]
\label{thm:joint-stopped-vector}
For every $Q\ge3$, odd or even, if $\{\log _2n\}\to t$, then
\[
 {1\over\sqrt n}(L_n-\mathbb EL_n,A_n,B_n)
 \Longrightarrow N_3(0,\Sigma_Q(t)),\qquad
 \Sigma_Q(t)=\operatorname {diag}(v_L(t),\sigma_Q^2,\sigma_Q^2).
\]
Moreover the normalized covariance matrices converge entry by entry to
$\Sigma_Q(t)$.  Consequently
\[
 {\operatorname {Var}V_{2/Q,n}\over n}\longrightarrow
 Q^2v_L(t)+2\sigma_Q^2,
\]
and the same scalar limit theorem follows from \eqref{eq:stopped-vector-id}.
\end{theorem}
\begin{proof}
We give all centering, bracket, exceptional-event and moment details.  Let
$\ell=\lfloor\frac12\log _2n\rfloor$, $K=2^\ell$, let $I_1,\ldots,I_K$
be the depth-$\ell$ cells, and let $N_I$ be their labelled occupancies.  Put
\[
 g_\ell=\mathbb E(g_Q\mid X(1),\ldots,X(\ell)),\quad
 \rho_\ell=g_Q-g_\ell,\quad M_{n,\ell}=\sum_i g_\ell(X_i),\quad
 R_{n,\ell}=\sum_i\rho_\ell(X_i).
\]
Thus $A_n=M_{n,\ell}+R_{n,\ell}$ and
$\|R_{n,\ell}\|_2/\sqrt n=\|\rho_\ell\|_2\to0$.  On the event
$E_{n,\ell}$ that no depth-$\ell$ cell is a singleton, every path reaches
the cut and, pathwise,
\begin{equation}
 L_n=n\ell+\sum_I L_{N_I}^{I},\qquad
 B_n=\sum_I G_{N_I}^{Y,I},\quad G_k^Y=\sum_{j=1}^kg_Q(Y_j).
                                                        \label{eq:joint-cut}
\end{equation}

Here is the promised martingale representation.  Reveal first the labelled
$\ell$-bit words, one label at a time, and then reveal, cell by cell, the
entire unused suffix blocks.  Denote the resulting filtration by
$\mathcal H_0\subset\cdots\subset\mathcal H_{n+K}$; in particular
$\mathcal H_n=\mathcal F_\ell$, the labelled allocation field.  For a fixed
$\lambda=(a,b,c)\in\mathbb R^3$ its differences are
\begin{align*}
 D_j&=b\,g_\ell(X_j),&&1\le j\le n,\\
 D_{n+r}&=a\{L_{N_{I_r}}^{I_r}-m(N_{I_r})\}
              +cG_{N_{I_r}}^{Y,I_r},&&1\le r\le K,
                                                        \label{eq:joint-diffs}
\end{align*}
where $m(k)=\mathbb EL_k$.  The first differences are adapted and centered
because the prefix words are iid and $\mathbb Eg_\ell=0$; conditional
Jensen gives their square integrability.  The second differences are
adapted.  Conditional on $\mathcal F_\ell$, PR~62's labelled cylinder
factorization, applied separately in every cell, makes the cell suffix
samples independent canonical samples.  Hence they are conditionally
centered and square integrable.  Thus \eqref{eq:joint-diffs}, rather than the
inapplicable identities $[A,A]=L_n$ or $[A,B]=L_n-n$, is the exact
Cram\'er--Wold martingale array.

For completeness, compute every bracket term.  The stopped-prefix field of
a canonical $k$-sample determines $L_k$, while the cylinder factorization
gives
\[
 \mathbb E(G_k^Y\mid\mathcal G_k)=0,\quad
 \mathbb E((G_k^Y)^2\mid\mathcal G_k)=k\sigma_Q^2.
\]
Consequently, term by term,
\[
 \operatorname {Var}(L_k-m(k))=\operatorname {Var}L_k,\quad
 \operatorname {Cov}(L_k,G_k^Y)=0,\quad
 \operatorname {Var}G_k^Y=k\sigma_Q^2.
\]
Different cells have zero conditional cross-brackets, as do the two reveal
stages.  The predictable bracket of \eqref{eq:joint-diffs} is therefore
\begin{equation}
 \langle S_\lambda\rangle
 =b^2n\|g_\ell\|_2^2+a^2\sum_I\operatorname {Var}L_{N_I}
       +c^2n\sigma_Q^2.                              \label{eq:joint-bracket}
\end{equation}
Write $v(k)=\operatorname {Var}L_k$.  The fixed-size trie renewal already
proved above gives $v(k)/k\to v_L(s)$ uniformly when
$\{\log _2k\}\to s$, and $v(k)\le Ck$.  A multinomial Chernoff union
bound gives $\max_I|N_I/(n/K)-1|=o_{\mathbb P}(1)$.  Since $\ell$ is an
integer, $\{\log _2(n/K)\}=\{\log _2n\}\to t$; hence, on that event,
\[
 {1\over n}\sum_Iv(N_I)
 =\sum_I{N_I\over n}{v(N_I)\over N_I}
 \longrightarrow v_L(t).
\]
(The empty terms are zero.)  This also follows directly from total variance:
conditional suffix variance is $\sum_Iv(N_I)$, while the conditional-mean
variance is $\operatorname {Var}\sum_Im(N_I)=O(K)=o(n)$ by
\eqref{eq:actual-occ}; the exceptional cut error below is $o_{L^2}(\sqrt n)$.
The ratio is bounded by $C$ because $v(k)\le Ck$, so convergence in
probability upgrades to $L^2$.
Together with $\|g_\ell\|_2^2\to\sigma_Q^2$, \eqref{eq:joint-bracket}
converges, after division by $n$, to
$a^2v_L(t)+(b^2+c^2)\sigma_Q^2$.

We now prove, rather than assume, the occupancy centering used here.  Direct
survival counting gives
\begin{equation}
 m(k)=k\sum_{r\ge0}\{1-(1-2^{-r})^{k-1}\},\qquad m(0)=m(1)=0. \label{eq:m-actual}
\end{equation}
This is an actual increment estimate, as follows.  Put $z=2^{-r}$ and
$q=1-z$.  The $r$th summand is $k(1-q^{k-1})$, and direct subtraction gives
\[
 \Delta^2\{k(1-q^{k-1})\}
 =q^{k-1}\{2zq-kz^2\}.
\]
Therefore
\[
 |\Delta^2m(k)|\le\sum_{r\ge0}
 \{2z(1-z)^k+kz^2(1-z)^{k-1}\}.
\]
For $z\le1/k$ the dyadic geometric sums are $O(1/k)$; for $z>1/k$,
$(1-z)^{k-1}\le e^{-(k-1)z}$ and the dyadic sum, after $u=kz$, is again
$O(1/k)$.  Thus, for $d(s)=m(s+1)-m(s)$,
\begin{equation}|d(s+1)-d(s)|\le {C\over s+1}.          \label{eq:difference-d}
\end{equation}
Replace one labelled cell choice by an independent choice.  If it moves a
label from occupancy $u$ to occupancy $w$, the actual change of
$S=\sum_I m(N_I)$ is $d(w)-d(u-1)$ (and is zero if the cells agree).  Write
$\lambda_0=n/K$ and $z_0=\lfloor\lambda_0\rfloor$.  For a binomial
occupancy $Z$, on $Z\ge z_0/2$, \eqref{eq:difference-d} gives
\[
 |d(Z)-d(z_0)|\le {C|Z-z_0|\over z_0/2+1},\qquad
 \mathbb E|d(Z)-d(z_0)|^2=O(1/\lambda_0).
\]
On $Z<z_0/2$, telescoping gives at most
$C\log((z_0+1)/(Z+1))$ and Chernoff's bound makes its squared expectation
$O(\log^2(z_0+1)e^{-c\lambda_0})=O(1/\lambda_0)$.  The size-biased source
occupancy has the same calculation after subtracting the moved label.
Thus, without assuming independence of the two affected cells,
\[
 \mathbb E(S-S^{(i)})^2\le C/\lambda_0=CK/n.
\]
Efron--Stein now yields
\begin{equation}
 \left\|\sum_I m(N_I)-\mathbb E\sum_I m(N_I)\right\|_2
 \le C\sqrt K=O(2^{\ell/2}).                         \label{eq:actual-occ}
\end{equation}
This calculation uses the actual mean \eqref{eq:m-actual}, and its constant
is uniform throughout $K\le C_0\sqrt n$.

The conditional Lyapunov condition is also explicit.  Set
$X_k=L_k-m(k)$ and
$c_k=k+m(B_k)+m(k-B_k)-m(k)$, where $B_k$ is binomial.  Conditional on
$B_k$, the child tries are independent, so
\[
 \mathbb E|X_{B_k}^{(0)}+X_{k-B_k}^{(1)}|^4
 =2\mathbb EM_4(B_k)+6\mathbb E\{v(B_k)v(k-B_k)\}.
\]
The displayed second-difference estimate and binomial fourth moments give
$\mathbb E|c_k|^4=O(k^2)$, while $v(k)=O(k)$.  Young's inequality gives
$M_4(k)\le(1+\epsilon)2\mathbb EM_4(B_k)+C_\epsilon k^2$.
After moving the self-atom $2(1+\epsilon)2^{-k}M_4(k)$ to the left, induction
closes because $2\mathbb E(B_k+1)^2/(k+1)^2\to1/2$.  Enlarging the constant
for finitely many $k$ proves
$\mathbb E|L_k-m(k)|^4\le C(k+1)^2$.

For the endpoint, suffix factorization gives the exact iid formula
\[
 \mathbb E|G_k^Y|^4
 =k\mathbb E|g_Q|^4+3k(k-1)\sigma_Q^4=O_Q((k+1)^2),
\]
including the exceptional singleton $G_1^Y=g_Q(Y_1)$.  Thus
$\mathbb E(|D_{n+r}|^4\mid\mathcal F_\ell)\le C_{\lambda,Q}(N_{I_r}+1)^2$.  No undefined
central window is needed: multinomial moments give deterministically
\[
 \mathbb E\sum_I(N_I+1)^2
 ={n^2\over K}+n(1-K^{-1})+2n+K,
\]
so Markov's inequality yields
\[
 {1\over n^2}\sum_I
 \mathbb E(|D_{n+I}|^4\mid\mathcal F_\ell)
 =O_{\mathbb P}(K^{-1}+n^{-1})=o_{\mathbb P}(1).
\]
For the prefix row, independence from $\mathcal H_{j-1}$ and conditional
Jensen give exactly
\[
 n^{-2}\sum_{j\le n}\mathbb E(|D_j|^4\mid\mathcal H_{j-1})
 ={b^4\mathbb E|g_\ell|^4\over n}
 \le {b^4\mathbb E|g_Q|^4\over n}=o(1).
\]
These conditional Lyapunov bounds imply conditional Lindeberg.  They prove
the scalar martingale CLT when the limiting bracket is positive; when it is
zero, \eqref{eq:joint-bracket} and conditional Chebyshev give convergence
to zero, so no division by a vanishing variance occurs.

It remains to justify the cut and all endpoint covariances in second
moment.  A labelled union bound gives
\[
 \mathbb P(E_{n,\ell}^c)\le n(1-K^{-1})^{n-1},
\]
which is superpolynomially small.  The fourth-moment bounds just displayed,
Minkowski, and $\mathbb E|A_n|^4+\mathbb E|B_n|^4=O_Q(n^2)$ show by
Cauchy--Schwarz that deleting $E_{n,\ell}^c$ changes each normalized
coordinate in $L^2$ by $o(1)$.  This is the exceptional-singleton
fourth-moment transfer; it applies to every linear combination, including a
combination whose Gaussian variance is zero, and supplies uniform
integrability of all normalized products.

The suffix factorization gives exactly
\begin{equation}\operatorname {Cov}(L_n,B_n)=0.        \label{eq:LBexact}\end{equation}
Indeed $L_n$ is $\mathcal G_n$-measurable and
$\mathbb E(B_n\mid\mathcal G_n)=0$.  Also
$\operatorname {Cov}(L_n,A_n)=0$ exactly: conditional on $X_i=x$, bitwise
XOR of every other string with $x$ preserves all prefix-equality relations,
so the conditional law of $L_n$ does not depend on $x$; summing
$\mathbb E[(L_n-\mathbb EL_n)g_Q(X_i)]$ proves the claim.  Finally, on
$E_{n,\ell}$ the block $B_n$ is conditionally centered given
$\mathcal F_\ell$, and is therefore orthogonal to $M_{n,\ell}$.  Hence
\[
 |\operatorname {Cov}(A_n,B_n)|
 \le\|R_{n,\ell}\|_2\|B_n\|_2+o(n)=o(n),
\]
because $\|B_n\|_2=\sqrt n\sigma_Q$; the displayed fourth-moment transfer
is the $o(n)$ exceptional term.  These are actual covariance estimates,
not consequences inferred from convergence in law.  Likewise
\eqref{eq:actual-occ} makes the conditional/global centering discrepancy
$o_{L^2}(\sqrt n)$.

Thus the martingale sum differs in $L^2/\sqrt n$ from the exact
Cram\'er--Wold statistic
$a(L_n-\mathbb EL_n)+bA_n+cB_n$ by a quantity tending to zero.  The scalar
CLT for every $(a,b,c)$ and Cram\'er--Wold prove the asserted joint limit.
The exact variance formulas, the three cross-covariance calculations, and
uniform integrability prove entrywise covariance convergence.

Nothing in this proof uses a Perron sign or an odd/even switching grid: it
uses only the fair trie and the certified coboundary
$F_{2/Q}-Q=g_Q-g_Q\circ T$ with $g_Q\in L^4$.  It therefore proves the
identical theorem separately for odd and even $Q$.  Parity first enters the
later discrepancy mixed forcing, which is not an ingredient of this theorem.
\end{proof}

\subsection{Ingredient 3: random occupancies and the actual block matrix}
For later use put
\[
 W_m=QL_m-\sum_{i=1}^m g_Q(Y_i),\qquad
 q_Q(m)=\operatorname {Var}\Delta_{Q,m},
\]
and define the two mixed coefficients
\[
 h_Q(m)=Q\operatorname {Cov}(L_m,\Delta_{Q,m}),\qquad
 p_Q(m)=\operatorname {Cov}\!\left(\sum_{i=1}^m
 g_Q(Y_i),\Delta_{Q,m}\right).
\]
The empty and singleton conventions are the actual ones: $L_0=L_1=0$,
$\Delta_{Q,0}=\Delta_{Q,1}=0$, and the endpoint sum is empty for $m=0$.
In particular, no singleton correction is suppressed in the following exact
identity:
\begin{equation}
 C_Q(m):=\operatorname {Cov}\binom{W_m}{\Delta_{Q,m}}
 =\begin{pmatrix}
 Q^2\operatorname {Var}L_m+m\sigma_Q^2&h_Q(m)-p_Q(m)\\
 h_Q(m)-p_Q(m)&q_Q(m)
 \end{pmatrix}.                                      \label{eq:actual-block-matrix}
\end{equation}
Indeed Lemma~\ref{lem:res-suffix-factor} gives
$\operatorname {Cov}(L_m,\sum_i g_Q(Y_i))=0$ and variance
$m\sigma_Q^2$; bilinearity gives the off-diagonal entry.  Notice that
$W_m$, rather than the complete stopped cost, occurs here.

We record the random-occupancy passage in the precise strength needed below.
This separates a genuine renewal obligation from a sometimes implicit
appeal to pointwise phase convergence.

\begin{lemma}[Uniform phase tails imply the block occupancy limit]
\label{lem:block-occupancy-transfer}
Suppose that continuous one-periodic functions $c_{11},c_{12},c_{22}$
satisfy, entrywise,
\begin{equation}
 \sup_{m\ge M}\left|m^{-1}C_Q(m)_{ab}
 -c_{ab}(\{\log _2m\})\right|\longrightarrow0
 \quad(M\to\infty).                                  \label{eq:block-uniform-tail}
\end{equation}
If $K=2^\ell\le C_0\sqrt n$, $n/K\to\infty$, and
$(N_I)_{I\in\mathcal D_\ell}$ are the labelled multinomial occupancies,
then, whenever $\{\log _2n\}\to t$,
\[
 {1\over n}\sum_{I\in\mathcal D_\ell}C_Q(N_I)
 \longrightarrow_{\mathbb P}
 \begin{pmatrix}c_{11}(t)&c_{12}(t)\\c_{12}(t)&c_{22}(t)\end{pmatrix}.
\]
Moreover, for any scalar sequence $u(m)$, a continuous phasewise limit
$u(m)/m\to c(\{\log _2m\})$ along every convergent phase sequence already
implies the uniform tail assertion analogous to
\eqref{eq:block-uniform-tail}.
\end{lemma}
\begin{proof}
Write $\lambda=n/K$.  Since $\ell$ is integral,
$\{\log _2\lambda\}=\{\log _2n\}$.  A binomial Chernoff bound and a union
bound give
\[
 \max_I|N_I-\lambda|=O_{\mathbb P}(\sqrt{\lambda\log K})
 =o_{\mathbb P}(\lambda).
\]
Thus, with probability tending to one, all $N_I$ tend uniformly to infinity
and
$\max_I|\log _2N_I-\log _2\lambda|\to0$.  Apply
\eqref{eq:block-uniform-tail} and uniform continuity on the phase circle,
entry by entry, to obtain
\[
 C_Q(N_I)_{ab}=N_I\{c_{ab}(t)+o_{\mathbb P}(1)\}
\]
uniformly in $I$.  Summing and using $\sum_I N_I=n$ proves the matrix
claim.  For the final assertion, if uniformity failed, there would be
$m_j\to\infty$ with an error bounded away from zero.  Compactness of the
phase circle supplies a subsequence with $\{\log _2m_j\}\to s$; the assumed
phasewise convergence and continuity at $s$ give a contradiction.
\end{proof}

For the $(1,1)$ entry, Theorem~\ref{thm:joint-stopped-vector} supplies

\]
For the $(2,2)$ entry the signed discrepancy renewal must supply the same

\subsection{Ingredient 4: the mixed covariance and its phase function}
\label{sec:mixed-kappa-d}
We now discharge the mixed input left open above.  Because two earlier modules
used the letter $h$ for different covariances, we keep both names visible.
Put, for a canonical $m$-sample,
\[
 G_{f,m}=\sum_{i=1}^m f(X_i),\qquad
 \widetilde h_Q(m)=\Cov(V_{2/Q,m},\Delta_{Q,m}),\qquad
 \widetilde p_Q(m)=\Cov(G_{g_Q,m},\Delta_{Q,m}).
\]
Thus $h_Q,p_Q$ in \eqref{eq:actual-block-matrix} use respectively the
LISTER length and the \emph{terminal} suffixes $Y_i$, whereas the tilded
quantities use the complete stopped cost and the \emph{entrance} points
$X_i$.  The stopped identity \eqref{eq:res-stop-id} gives the exact bridge
\begin{equation}
 C_Q(m)_{12}=h_Q(m)-p_Q(m)
 =\widetilde h_Q(m)-\widetilde p_Q(m).                 \label{eq:mixed-bridge-d}
\end{equation}
This identity, rather than an identification of the two notationally
different $h$'s or $p$'s, is what connects the renewal below to Ingredient~3.

Write $a(m)=\E\Delta_{Q,m}$, $v(m)=\E V_{2/Q,m}$ and abbreviate
$\bar D_m=\bar D_{2/Q,m}$.  If $J$ is the labelled left occupancy, let
\[
 A=V_{2/Q,J}^{(0)}+V_{2/Q,m-J}^{(1)},\quad
 B=\Delta_{Q,J}^{(0)}+\Delta_{Q,m-J}^{(1)},\quad
 \bar\Phi_m={\bf1}_{\{m\ge2\}}\sum_{i=1}^mF_{2/Q}(X_i).
\]
Conditionally on all labelled first bits the two child samples are independent
canonical samples, but $\bar D_m$ is a function of both of them.  The exact
root recursions are
\[
 V_{2/Q,m}=A+\bar\Phi_m,
 \qquad \Delta_{Q,m}=B+\bar D_m.                       \label{eq:mixed-root-d}
\]
The barred toll is essential: both variables and all covariances below vanish
at $m=0,1$.

\begin{proposition}[Exact four-term mixed forcing]\label{prop:mixed-force-d}
For $B_m\sim\operatorname {Bin}(m,1/2)$,
\begin{equation}
 \widetilde h_Q(m)=2\E\widetilde h_Q(B_m)+s_Q(m),       \label{eq:mixed-renew-d}
\end{equation}
where $s_Q(0)=s_Q(1)=0$ and, without changing any sign,
\begin{align}
 s_Q(m)={}&\Cov\{v(J)+v(m-J),a(J)+a(m-J)\}             \label{eq:mixed-four-d}\\
 &+\Cov(A,\bar D_m)+\Cov(\bar\Phi_m,B)
   +\Cov(\bar\Phi_m,\bar D_m).\notag
\end{align}
Moreover, for every $\epsilon>0$,
\begin{equation}|s_Q(m)|\le C_{Q,\epsilon}(m+1)^{1/2+\epsilon}.\label{eq:mixed-force-bound-d}\end{equation}
\end{proposition}
\begin{proof}
Apply total covariance to \eqref{eq:mixed-root-d}.  Conditional on $J$, the
centered child pairs are independent across branches.  Their two same-branch
covariances give $\widetilde h_Q(J)+\widetilde h_Q(m-J)$; their conditional
means give the first covariance in \eqref{eq:mixed-four-d}.  Expanding the
remaining covariance of $A+\bar\Phi_m$ with $B+\bar D_m$ gives, in order,
the other three displayed covariances.  Binomial reflection turns the two
same-branch terms into $2\E\widetilde h_Q(B_m)$.  This proves the signed
identity; in particular neither child/root covariance has been declared zero.

We next bound its four terms without using a bound on $\widetilde h_Q$.
The estimates \eqref{eq:zreg-d} and \eqref{eq:pair-d}, applied to $a$, say
\[
 |\nabla a(u)-\nabla a(w)|\le C_{Q,\beta}|u-w|
 (1+\min(u,w))^{-\beta},\qquad \sup_u|\nabla a(u)|<\infty.
\]
Taking expectations in the stopped identity \eqref{eq:res-stop-id}
and using $\E g_Q(X_i)=\E g_Q(Y_i)=0$ gives exactly
$v(m)=Q\E L_m$.  Formula \eqref{eq:m-actual} gives
$|\nabla\E L_u|\le C\log(u+2)$ by splitting its dyadic sum at
$2^r=u+1$, while \eqref{eq:difference-d} gives, by telescoping,
$|\nabla\E L_u-\nabla\E L_w|\le
 C|u-w|/(1+\min(u,w))$.  Hence
$|\nabla v(u)|\le C_Q\log(u+2)$ and
$|\nabla v(u)-\nabla v(w)|\le
 C_Q|u-w|/(1+\min(u,w))$.
Subtract the two midpoint values in the first covariance.  On
$|J-m/2|\le m/4$, midpoint pairing (with the two middle integers retained
when $m$ is odd) and the preceding estimates bound the two centered factors
by respectively
\[
 C_Q(1+|J-m/2|)^2/m,\qquad
 C_{Q,\beta}(1+|J-m/2|)^2m^{-\beta}.
\]
Binomial eighth moments are $O(m^4)$; on the complementary event the bounded
increments and the Chernoff bound absorb every polynomial and logarithm.
Taking $\beta$ arbitrarily close to one (and weakening the resulting power)
therefore gives
\begin{equation}
 |\Cov\{v(J)+v(m-J),a(J)+a(m-J)\}|
 \le C_{Q,\epsilon}m^{1/2+\epsilon}.                  \label{eq:childmeans-mix-d}
\end{equation}
This deliberately records the weaker summable exponent needed below.

The stopped second-moment theorem gives $\Var A=O_Q(m)$, while Proposition
\ref{prop:defect-moments-d} gives $\sup_m\Var\bar D_m<\infty$.
Cauchy--Schwarz therefore bounds $|\Cov(A,\bar D_m)|$ by $C_Q\sqrt m$.
For the last term the constant $Qm$ drops out and
$F_{2/Q}-Q=g_Q-g_Q\circ T$; hence another Cauchy--Schwarz estimate gives
\begin{equation}
 |\Cov(\bar\Phi_m,\bar D_m)|
 \le\sqrt{m}\,\|g_Q-g_Q\circ T\|_2
       \sup_r\|\bar D_r-\E\bar D_r\|_2.               \label{eq:rootroot-mix-d}
\end{equation}

It remains to control $\Cov(\bar\Phi_m,B)$, the term where an unsigned
domination would lose the Perron cancellation.  For every $f\in L^2$ put
\[
 p_f(m)=\Cov(G_{f,m},\Delta_{Q,m}),\qquad
 \rho_f(m)=\Cov(G_{f,m},\bar D_m).
\]
These definitions depend only on $f-\E f$, because constants have zero
covariance with either centered second argument; thus they also apply to the
not necessarily centered branch restrictions used below.
Splitting each $X_i=(b+\xi_i)/2$ by its labelled first bit and retaining the
conditional means gives exactly
\begin{equation}p_f(m)=2\E p_{Pf}(B_m)+\rho_f(m).       \label{eq:pf-system-d}\end{equation}
Indeed the same-child centered terms give $P f$, the cross-child terms vanish
by conditional independence, and the remaining conditional-mean covariance
is odd in $J-m/2$ whereas $a(J)+a(m-J)$ is even.  Thus it cancels by
$J\leftrightarrow m-J$, not by independence.  Also
\begin{equation}|\rho_f(m)|\le C_Q\sqrt m\,\|f\|_2.    \label{eq:rho-f-d}\end{equation}

Here are the two parity systems, now based on the explicit calculation in
Lemma~\ref{lem:explicit-perron-parity-d}.  Put $L=L_Q=R_Q+1$; zero vectors
at the end may harmlessly be deleted.  For odd $Q$, let
$W_Q=Pg_Q-g_Q/2=-K_Q/2$.  Then $P^LW_Q=0$, and in the ordered list
$e_0=g_Q,e_1=W_Q,e_2=PW_Q,\ldots,e_L=P^{L-1}W_Q$,
\eqref{eq:pf-system-d} reads
\begin{align*}
 p_{e_0}(m)&=\E p_{e_0}(B_m)+2\E p_{e_1}(B_m)+\rho_{e_0}(m),\\
 p_{e_i}(m)&=2\E p_{e_{i+1}}(B_m)+\rho_{e_i}(m)\quad(1\le i<L),\\
 p_{e_L}(m)&=\rho_{e_L}(m),
\end{align*}
and
\begin{equation}u_Q:=2P(F_{2/Q}-Q)=-g_Q+2W_Q.          \label{eq:odd-sign-d}\end{equation}
For even $Q$, put $K_Q=g_Q-2Pg_Q$, which
Lemma~\ref{lem:explicit-perron-parity-d} identifies with the finite sum
\eqref{eq:g-explicit-d}; directly, $P^LK_Q=0$.  In the list
$e_0=g_Q,e_1=K_Q,e_2=PK_Q,\ldots,e_L=P^{L-1}K_Q$, the system is instead
\begin{align*}
 p_{e_0}(m)&=\E p_{e_0}(B_m)-\E p_{e_1}(B_m)+\rho_{e_0}(m),\\
 p_{e_i}(m)&=2\E p_{e_{i+1}}(B_m)+\rho_{e_i}(m)\quad(1\le i<L),\\
 p_{e_L}(m)&=\rho_{e_L}(m),
\end{align*}
and
\begin{equation}u_Q=2P(F_{2/Q}-Q)=-g_Q-K_Q.             \label{eq:even-sign-d}\end{equation}
Thus the odd $+2W_Q$ and even $-K_Q$ signs have not been conflated.

Backward induction using \eqref{eq:rho-f-d} and
$\E\sqrt{B_m+1}\le\sqrt{m/2+1}$ gives $p_{e_i}(m)=O_Q(\sqrt{m+1})$
for $i\ge1$.  For $p_{e_0}$, isolate the self-atom $B_m=m$ and use the
strict square-root contraction
$\E\sqrt{B_m+1}/\sqrt{m+1}<1$ after finitely many $m$; enlarging the
constant over that finite set closes strong induction.  In both parities,
\begin{equation}|p_{g_Q}(m)|+|p_{u_Q}(m)|\le C_Q\sqrt{m+1}.\label{eq:p-sqrt-d}\end{equation}

For completeness, expose all four entrance/terminal signs.  If
$g_b(x)=g_Q((b+x)/2)$, conditional first-bit expansion gives
\begin{align*}
 \Cov(\bar\Phi_m,B)=\E[&p_{g_0}(J)-p_{g_Q}(J)
 +p_{g_1}(m-J)-p_{g_Q}(m-J)]\\
 &=\E p_{2Pg_Q-2g_Q}(B_m)=\E p_{u_Q}(B_m).             \label{eq:four-endpoints-d}
\end{align*}
The first and third terms are the two entrance covariances and the second
and fourth are the two terminal covariances.  Cross-branch covariances vanish
only after conditioning; the occupancy-mean term cancels by the odd/even
reflection just used.  Formula \eqref{eq:p-sqrt-d} bounds this term by
$C_Q\sqrt{m+1}$.  Combining this with
\eqref{eq:childmeans-mix-d}, \eqref{eq:rootroot-mix-d}, and the bound on
$\Cov(A,\bar D_m)$ proves \eqref{eq:mixed-force-bound-d} noncircularly.
\end{proof}

\begin{theorem}[Signed renewal, explicit $\kappa_Q$, and uniform phase tails]
\label{thm:mixed-kappa-d}
Define, for $t\in\mathbb R$,
\begin{align}
 \kappa_Q(t)
 &:=\sum_{j\in\mathbb Z}{\E s_Q(\operatorname {Pois}(2^{t-j}))\over2^{t-j}}
                                                               \label{eq:kappa-pois-d}\\
 &=\sum_{m\ge2}{s_Q(m)\over m!}
   \sum_{j\in\mathbb Z}(2^{t-j})^{m-1}e^{-2^{t-j}}.    \label{eq:kappa-series-d}
\end{align}
The series converge normally on the phase circle.  Consequently $\kappa_Q$
is finite, bounded, continuous and one-periodic.  Moreover
\begin{equation}
 {\widetilde h_Q(m)\over m}\longrightarrow\kappa_Q(t)
 \quad\hbox{whenever }\{\log_2m\}\to t,               \label{eq:h-phase-d}
\end{equation}
and in fact
\begin{equation}
 \sup_{m\ge M}\left|{\widetilde h_Q(m)\over m}
       -\kappa_Q(\{\log_2m\})\right|\longrightarrow0. \label{eq:h-uniform-phase-d}
\end{equation}
The same conclusions hold for the actual block entry $C_Q(m)_{12}$.
\end{theorem}
\begin{proof}
Fix $\alpha\in(1/2,1)$ in \eqref{eq:mixed-force-bound-d}.  Iterating
\eqref{eq:mixed-renew-d} through $R$ independent thinnings gives
\begin{equation}
 \widetilde h_Q(m)=\sum_{r<R}2^r\E s_Q(\operatorname {Bin}(m,2^{-r}))
 +2^R\E\widetilde h_Q(\operatorname {Bin}(m,2^{-R})). \label{eq:h-iterate-d}
\end{equation}
For fixed $m$ the last term tends to zero.  Indeed
$\widetilde h_Q(0)=\widetilde h_Q(1)=0$ and
$\Pr\{\operatorname {Bin}(m,p)\ge2\}\le m(m-1)p^2/2$;
the maximum of $|\widetilde h_Q(k)|$ over $2\le k\le m$ is finite.  This
proves the boundary limit without assuming $\widetilde h_Q=O(m)$.

The resulting infinite sum is absolutely convergent.  With
$\lambda=m2^{-r}$, its normalized summand is bounded, uniformly in $m$, by
\begin{equation}
 {\E|s_Q(\operatorname {Bin}(m,\lambda/m))|\over\lambda}
 \le C_{Q,\alpha}\begin{cases}\lambda,&0<\lambda\le1,\\
 \lambda^{\alpha-1},&\lambda\ge1.
 \end{cases}                                           \label{eq:renew-majorant-d}
\end{equation}
For the first line use $s_Q(0)=s_Q(1)=0$ and
$k^\alpha\le k(k-1)$ for $k\ge2$; for the second use concavity,
$\E B^\alpha\le(\E B)^\alpha$.  Thus both dyadic tails are geometric.

Write $m=2^{M_m+t_m}$ with $M_m=\lfloor\log_2m\rfloor$ and $t_m\in[0,1)$,
and reindex $j=r-M_m$.  For fixed $j$, the binomial variable converges to
$\operatorname {Pois}(2^{t-j})$ when $t_m\to t$; its $\alpha$-moments are
uniformly integrable, so the corresponding expectation converges.  The
majorant \eqref{eq:renew-majorant-d} permits passage through the bilateral
sum and proves \eqref{eq:kappa-pois-d} and \eqref{eq:h-phase-d}.
Expanding the Poisson expectation and using the same absolute majorant gives
\eqref{eq:kappa-series-d}.

More explicitly, uniformly for $t\in[0,1]$, the terms with $j>R$ are at
most $C_Q\sum_{j>R}2^{-j}$, while those with $j<-R$ are at most
$C_Q\sum_{j<-R}2^{-(1-\alpha)|j|}$.  This is the required uniform phase-tail
estimate, not merely pointwise convergence.  The Weierstrass test and
continuity of each Poisson term prove continuity and boundedness; replacing
$t$ by $t+1$ and reindexing $j$ proves one-periodicity.  The same uniform
two-tail argument shows that convergence along every phase sequence is
uniform in the compactness sense \eqref{eq:h-uniform-phase-d} (otherwise a
counterexample sequence has a convergent phase subsequence).

Finally \eqref{eq:p-sqrt-d} gives
$|\widetilde p_Q(m)|/m\le C_Q/\sqrt{m+1}$.  By
\eqref{eq:mixed-bridge-d},
\begin{equation}
 {C_Q(m)_{12}\over m}={\widetilde h_Q(m)-\widetilde p_Q(m)\over m}
 \longrightarrow\kappa_Q(t),                           \label{eq:block-mixed-phase-d}
\end{equation}
with the uniform tail \eqref{eq:block-uniform-tail}.  Also, for $K$ cells,
$n^{-1}\sum_I|\widetilde p_Q(N_I)|\le C_Q\sqrt{K/n}$ by
Cauchy--Schwarz.  Hence the terminal/entrance correction is controlled both
at fixed size and after the half-log multinomial cut.  This proves the exact
mixed hypothesis consumed by Lemma~\ref{lem:block-occupancy-transfer}; no
convergence-in-law assertion has been used as covariance convergence.
\end{proof}

\begin{proposition}[upper-discrepancy residual]
\label{prop:halfres-d}
For fixed $Q\ge3$, all $n,\ell\ge0$ satisfy
\begin{equation}
 \|R_{n,\ell}-\mathbb ER_{n,\ell}\|_2\le C_Q2^{\ell/2},
 \qquad \operatorname {Var}R_{n,\ell}\le C_Q2^\ell.     \label{eq:halfres-bound-d}
\end{equation}
In particular, for
$\ell_n=\lfloor\frac12\log_2n\rfloor$ this norm is
$O_Q(n^{1/4})=o(\sqrt n)$.
\end{proposition}
\begin{proof}
Fix any $\beta\in(1/2,1)$.  The already proved smoothing estimate
\eqref{eq:dreg-d}, including its explicitly checked finite initial range,
says
\begin{equation}
 |d_Q(k+1)-d_Q(k)|\le C_{Q,\beta}(k+1)^{-\beta}\quad(k\ge0).       \label{eq:halfres-diff-d}
\end{equation}
At level $r$ set
\begin{equation}
 F_{r,n}:=\sum_{I\in\mathcal D_r}\{\bar D_Q(S_I)-d_Q(N_I)\},
 \qquad M_{r,n}:=\sum_{I\in\mathcal D_r}d_Q(N_I),
 \qquad U_{r,n}=F_{r,n}+M_{r,n}.                         \label{eq:halfres-split-d}
\end{equation}
Let $\mathcal P_r$ be the sigma-field generated by the labelled allocation
of every point to its depth-$r$ cell.  Given $\mathcal P_r$, the binary
suffixes belonging to distinct cells are disjoint subfamilies of independent
bits.  After affine rescaling, the cell samples are therefore independent
canonical samples of the fixed sizes $N_I$.  Consequently
$\mathbb E(F_{r,n}\mid\mathcal P_r)=0$, and conditional independence and
Proposition~\ref{prop:defect-moments-d} give
\begin{equation}
 \mathbb E(F_{r,n}^2\mid\mathcal P_r)
 =\sum_{I\in\mathcal D_r}\operatorname {Var}(\bar D_{Q,N_I})
 \le C_Q2^r,
 \qquad \|F_{r,n}\|_2\le C_Q2^{r/2}.                    \label{eq:halfres-fluc-d}
\end{equation}

Here is the missing influence calculation for $M_{r,n}$.  Replace label
$X_i$ by an independent $X_i'$.  If their depth-$r$ cells coincide, the
statistic does not change.  Otherwise call those cells $A,B$, and let
$K_A,K_B$ be their occupancies among the other $n-1$ labels.  Exactly two
occupancies change, each by one, and
\[
 M_{r,n}-M_{r,n}^{(i)}
 =\{d_Q(K_A+1)-d_Q(K_A)\}
  -\{d_Q(K_B+1)-d_Q(K_B)\}.
\]
Each of $K_A,K_B$ has the marginal law
$K\sim\operatorname {Bin}(n-1,2^{-r})$ (no independence between them is
claimed or needed).  Thus \eqref{eq:halfres-diff-d}, $(x-y)^2\le2x^2+2y^2$,
and replacement Efron--Stein yield
\begin{equation}
 \operatorname {Var}M_{r,n}
 \le C_{Q,\beta}n\,\mathbb E(K+1)^{-2\beta}.             \label{eq:halfres-es-d}
\end{equation}
Put $\mu=(n-1)2^{-r}$.  If $\mu\ge1$, then on
$\{K\ge\mu/2\}$ the integrand is at most $C\mu^{-2\beta}$,
whereas Chernoff bounds the complementary probability by $e^{-c\mu}$.
Hence $\mathbb E(K+1)^{-2\beta}\le C_\beta\mu^{-2\beta}$.
Since $n\le2\mu2^r$ for $n\ge2$ and $1-2\beta<0$,
\eqref{eq:halfres-es-d} is at most $C_Q2^r$.  If $\mu\le1$, use the
elementary bound $(K+1)^{-2\beta}\le1$ and
$n\le2^r+1\le2^{r+1}$; the same conclusion follows.  The cases $n=0,1$
are immediate because every barred defect vanishes.  We have proved
\begin{equation}
 \|M_{r,n}-\mathbb EM_{r,n}\|_2\le C_Q2^{r/2}.            \label{eq:halfres-mean-d}
\end{equation}

Defects at different levels are not asserted to be independent or
orthogonal.  Instead, using $\mathbb EF_{r,n}=0$, the exact split is
\[
 R_{n,\ell}-\mathbb ER_{n,\ell}
 =\sum_{r<\ell}F_{r,n}
   +\sum_{r<\ell}\{M_{r,n}-\mathbb EM_{r,n}\}.
\]
Minkowski's inequality, \eqref{eq:halfres-fluc-d}, and
\eqref{eq:halfres-mean-d} now give
\[
 \|R_{n,\ell}-\mathbb ER_{n,\ell}\|_2
 \le C_Q\sum_{r<\ell}2^{r/2}\le C_Q2^{\ell/2}.
\]
Squaring proves the variance assertion.  At the literal cut,
$2^{\ell_n/2}\le n^{1/4}$, which proves the stated $o(\sqrt n)$ bound.
This proof uses only the root-defect moment bound, mean first-difference
regularity, binomial occupancy estimates, and Efron--Stein.  In particular it
uses neither the resonant CLT, any limiting variance phase, the present
residual estimate, nor the positivity argument.
\end{proof}

Write $r_Q(m)$ for the exact signed forcing in
\eqref{eq:signed-force-d}, so that
$q_Q(m)=2\mathbb E q_Q(\operatorname {Bin}(m,1/2))+r_Q(m)$ and
$r_Q(0)=r_Q(1)=0$.  The signed-renewal construction of the discrepancy
phase is the explicit normally convergent series
\begin{equation}
 \gamma_Q(t)=\sum_{m\ge2}{r_Q(m)\over m!}
 \sum_{j\in\mathbb Z}(2^{t-j})^{m-1}e^{-2^{t-j}}.       \label{eq:gamma-series-d}
\end{equation}
Indeed \eqref{eq:rsub-d} permits $|r_Q(m)|\le C_Qm^\alpha$ with
$\alpha<1$.  For $x=2^{t-j}$ the Poisson expectation form of this series
has the same two tails as in Theorem~\ref{thm:mixed-kappa-d}: for $x\le1$
the facts $r_Q(0)=r_Q(1)=0$ give a bound $C_Qx$, while for $x\ge1$ the
bound is $C_Qx^{\alpha-1}$.  Both dyadic tails are summable uniformly in
$t\in[0,1]$.  Thus the series is continuous and one-periodic, its renewal
boundary vanishes as in Theorem~\ref{thm:renew-d}, and compactness upgrades
phasewise convergence to
\begin{equation}
 \sup_{m\ge M}
 \left|{q_Q(m)\over m}-\gamma_Q(\{\log_2m\})\right|\longrightarrow0.
                                                               \label{eq:gamma-uniform-d}
\end{equation}
The mixed phase $\kappa_Q$ is the explicit, normally convergent series
\eqref{eq:kappa-pois-d}--\eqref{eq:kappa-series-d}.
Proposition~\ref{prop:mixed-force-d} retains its
child-mean, child/root, entrance/terminal, and root/root terms and proves the
shared-descendant estimate from the parity-specific finite Perron systems.
Then
\begin{equation}
 n^{-1}\operatorname{Cov}(V_{2/Q,n},\Delta_{2/Q,n})\to\kappa_Q(t),\qquad
 n^{-1}\operatorname{Var}\Delta_{2/Q,n}\to\gamma_Q(t).       \label{eq:res-phases-d}
\end{equation}
Normal convergence makes both phases continuous and one-periodic.

\subsection{Ingredient 5: adaptive fourth moments and the half-log block CLT}
\label{sec:ingredient-five-d}
Put $a(m)=\mathbb E\Delta_{Q,m}$, $Z_m=\Delta_{Q,m}-a(m)$,
$q(m)=\mathbb EZ_m^2$, and $M_4(m)=\mathbb E|Z_m|^4$.  We use the
literal cut
\[
 \ell_n=\left\lfloor\tfrac12\log _2n\right\rfloor,
 \qquad K_n=2^{\ell_n},\qquad
 \mathcal P_{\ell_n}=\sigma\{(i,X_i(1)\cdots X_i(\ell_n)):1\le i\le n\},
\]
and write $N_I$ for the occupancy of $I\in\mathcal D_{\ell_n}$.  Thus
$K_n\asymp\sqrt n$ and $\lambda_n=n/K_n\asymp\sqrt n$.  On the event
$E_{n,\ell_n}$ that no cut cell is a singleton, the exact lower-block
variables are
\[
 W_{N_I}^{I}=QL_{N_I}^{I}-\sum_{i\in I}g_Q(Y_i),\qquad
 \Delta_{Q,N_I}^{I},\qquad
 \mathcal Z_I=\binom{W_{N_I}^{I}-\mathbb EW_{N_I}}
 {\Delta_{Q,N_I}^{I}-a(N_I)}.                         \tag{5.1}\label{eq:i5-block}
\]
Here and below expectations indexed by $N_I$ mean expectation in a canonical
sample of that fixed size.  Lemma~\ref{lem:res-suffix-factor} says that,
conditionally on $\mathcal P_{\ell_n}$, the vectors in \eqref{eq:i5-block}
are independent and have exactly those canonical laws.  The variables above,
not complete stopped costs, are the conditional array.

\begin{lemma}[adaptive discrepancy fourth moment]\label{lem:i5-fourth}
For every $Q\ge3$, odd or even,
\begin{equation}
 \mathbb E|\Delta_{Q,m}-a(m)|^4\le C_Q(m+1)^2\qquad(m\ge0).       \tag{5.2}\label{eq:i5-delta-fourth}
\end{equation}
\end{lemma}
\begin{proof}
Let $J\sim\operatorname {Bin}(m,1/2)$.  The exact barred recursion
\eqref{eq:delta-rec-d}, including $m=0,1$ and empty children, gives
\begin{align}
 Z_m&=X+Y+H_m,\qquad X=Z_J^{(0)},\quad Y=Z_{m-J}^{(1)},             \tag{5.3}\label{eq:i5-centered-rec}\\
 H_m&=\bar D_{2/Q,m}+a(J)+a(m-J)-a(m).                              \tag{5.4}\label{eq:i5-H}
\end{align}
The root term is allowed to depend on both children.  If
$d(m)=\mathbb E\bar D_{2/Q,m}$ and $f_m(j)=a(j)+a(m-j)$, the exact mean
recurrence \eqref{eq:amean-exact-d} rewrites it as
\[
 H_m=\{\bar D_{2/Q,m}-d(m)\}+\{f_m(J)-\mathbb Ef_m(J)\}.           \tag{5.5}\label{eq:i5-Hcenter}
\]
Proposition~\ref{prop:defect-moments-d} bounds the fourth moment of the first
brace uniformly.  For the second, choose any $\beta\in(1/2,1)$.  The even
and odd midpoint identities preceding \eqref{eq:pair-d} give, on
$|J-m/2|\le m/4$,
\[
 |f_m(J)-c_m|\le C_\beta(1+|J-m/2|)^2m^{-\beta}.
\]
Since the eighth centered binomial moment is $O(m^4)$, its fourth moment is
$O(1+m^{4-4\beta})=O((m+1)^2)$ because $\beta>1/2$.  Off that event bounded first differences give
$|f_m(J)-c_m|\le C(m+1)$ and Chernoff gives probability $O(e^{-cm})$.
Finally $\|U-\mathbb EU\|_4\le2\|U-c\|_4$.  Enlarging the constant over
small $m$ therefore proves
\begin{equation}\mathbb E|H_m|^4\le C_Q(m+1)^2.                    \tag{5.6}\label{eq:i5-H4}\end{equation}
This argument explicitly uses the separate odd midpoint pair in
\eqref{eq:pair-d}; it does not assume that $\nabla a$ decays.

Conditionally on $J$, $X,Y$ are independent and centered, so the complete
child expansion is
\begin{equation}
 \mathbb E\{(X+Y)^4\mid J\}=M_4(J)+M_4(m-J)+6q(J)q(m-J).           \tag{5.7}\label{eq:i5-child4}
\end{equation}
Indeed the $X^3Y$ and $XY^3$ terms vanish by conditional centering, while the
$6X^2Y^2$ term displayed above does not vanish.  With $S=X+Y$, no term
involving $H_m$ may be discarded:
\begin{equation}
 M_4(m)=\mathbb ES^4+4\mathbb E(S^3H_m)+6\mathbb E(S^2H_m^2)
       +4\mathbb E(SH_m^3)+\mathbb EH_m^4.                         \tag{5.8}\label{eq:i5-full4}
\end{equation}
For $0<\epsilon\le1$, the three weighted Young inequalities
\[
4|S|^3|H|\le3\epsilon|S|^4+\epsilon^{-3}|H|^4,
\quad6|S|^2|H|^2\le3\epsilon|S|^4+3\epsilon^{-1}|H|^4,
\quad4|S||H|^3\le\epsilon|S|^4+3\epsilon^{-1/3}|H|^4
\]
control every mixed term.  By the independently proved linear bound
\eqref{eq:renew-result-d}, $q(j)\le C_Q(j+1)$.  Hence
\begin{equation}
 M_4(m)\le(1+7\epsilon)\,2\mathbb EM_4(B_m)+C_{Q,\epsilon}(m+1)^2. \tag{5.9}\label{eq:i5-rec4}
\end{equation}
This uses the second moment only to bound the retained product in
\eqref{eq:i5-child4}, never to infer a fourth moment.

We finish the strong induction, including its self-atoms.  Choose a fixed
$\epsilon>0$ so small that $(1+7\epsilon)/2<\rho<1$.  Since
\[
 {2\mathbb E(B_m+1)^2\over(m+1)^2}\longrightarrow\tfrac12,
\]
for all $m\ge m_0$ the same ratio multiplied by $1+7\epsilon$ is at most
$\rho$.  Before applying the induction hypothesis, remove $B_m=m$ from both
reflected child terms in \eqref{eq:i5-rec4}; the two atoms $J=0,m$ contribute
exactly $(1+7\epsilon)2^{1-m}M_4(m)$.  Thus
\[
 [1-(1+7\epsilon)2^{1-m}]M_4(m)
 \le (1+7\epsilon)2\mathbb E[M_4(B_m);B_m<m]+C_{Q,\epsilon}(m+1)^2.
\]
Increase $m_0$ so that the left coefficient and the contraction leave a
fixed positive margin.  Assuming $M_4(j)\le A(j+1)^2$ for $j<m$, choose $A$
large enough to absorb the last term in that margin, and then enlarge it over
the finite range $m<m_0$.  Strong induction proves \eqref{eq:i5-delta-fourth}.
The atoms at $0$ and $1$ are genuine: $Z_0=Z_1=0$; none was silently put
under the induction hypothesis.
\end{proof}

The stopped-block estimate proved in Theorem~\ref{thm:joint-stopped-vector}
and \eqref{eq:i5-delta-fourth}, together with
$|u+v|^4\le8(|u|^4+|v|^4)$, give, for every
$\lambda=(\lambda_1,\lambda_2)$,
\begin{equation}
 \mathbb E|\lambda_1(W_m-\mathbb EW_m)+\lambda_2Z_m|^4
 \le C_Q(|\lambda_1|+|\lambda_2|)^4(m+1)^2.             \tag{5.10}\label{eq:i5-cw4}
\end{equation}
The constant is therefore uniform for $\lambda$ in a compact set.

For fixed $\lambda$ define
$X_{I,n}(\lambda)=\lambda^{\mathsf T}\mathcal Z_I$.  Conditional on
$\mathcal P_{\ell_n}$ these variables are independent and centered, and
\begin{equation}
 \sum_I\mathbb E(|X_{I,n}(\lambda)|^4\mid\mathcal P_{\ell_n})
 \le C_{Q,\lambda}\sum_I(N_I+1)^2.                       \tag{5.11}\label{eq:i5-numerator}
\end{equation}
There is no replacement of occupancies by their mean.  Direct multinomial
calculation gives
\begin{equation}
 \mathbb E\sum_I(N_I+1)^2={n^2\over K_n}+n(1-K_n^{-1})+2n+K_n,
                                                                    \tag{5.12}\label{eq:i5-occ2}
\end{equation}
so Markov's inequality and $K_n\asymp\sqrt n$ give a numerator
$O_{\mathbb P}(n^{3/2})=o_{\mathbb P}(n^2)$.

Let
\[
 B_Q(t)=\begin{pmatrix}
 Q^2v_L(t)+\sigma_Q^2&\kappa_Q(t)\\
 \kappa_Q(t)&\gamma_Q(t)
 \end{pmatrix}.
\]
The conditional variance denominator is exactly
\begin{equation}
 S_n^2(\lambda)=\sum_I\operatorname {Var}(X_{I,n}(\lambda)\mid\mathcal P_{\ell_n})
 =\lambda^{\mathsf T}\!\left\{\sum_IC_Q(N_I)\right\}\lambda.    \tag{5.13}\label{eq:i5-denom}
\end{equation}
Ingredient~3's uniform-tail transfer, with the stopped entry from
Ingredient~2, the mixed entry $\kappa_Q$ from Ingredient~4, and the signed
variance phase $\gamma_Q$ from \eqref{eq:res-phases-d}, proves, whenever
$\{\log_2n\}\to t$,
\begin{equation}S_n^2(\lambda)/n\longrightarrow_{\mathbb P}
 v_\lambda(t):=\lambda^{\mathsf T}B_Q(t)\lambda.          \tag{5.14}\label{eq:i5-varlimit}\end{equation}
Notice that the block $(1,1)$ entry has only one $\sigma_Q^2$; the other is
the upper-prefix entrance projection below.  The matrix is positive
semidefinite because it is a limit of covariance matrices.  If
$v_\lambda(t)>0$, \eqref{eq:i5-numerator}--\eqref{eq:i5-varlimit} yield the
conditional Lyapunov ratio
\begin{equation}
 {\sum_I\mathbb E(|X_{I,n}(\lambda)|^4\mid\mathcal P_{\ell_n})
  \over S_n^4(\lambda)}=O_{\mathbb P}(n^{-1/2})\longrightarrow0.  \tag{5.15}\label{eq:i5-lyap}
\end{equation}
If $v_\lambda(t)=0$, we do not divide: conditional Chebyshev and
\eqref{eq:i5-varlimit} show $n^{-1/2}\sum_IX_{I,n}(\lambda)\to0$ in
conditional probability, in probability.  Explicitly, if $\det B_Q(t)>0$
only $\lambda=0$ is exceptional; if $\det B_Q(t)=0$, the null space is the
kernel of the displayed matrix (when its first diagonal entry is positive it
is spanned by $(-\kappa_Q(t),Q^2v_L(t)+\sigma_Q^2)$).

For positive directions, conditional Lyapunov implies conditional Lindeberg
because
\[
 {1\over S_n^2}\sum_I\mathbb E[X_{I,n}^2{\bf1}_{\{|X_{I,n}|>\varepsilon S_n\}}
 \mid\mathcal P_{\ell_n}]
 \le{1\over\varepsilon^2S_n^4}\sum_I\mathbb E(|X_{I,n}|^4\mid\mathcal P_{\ell_n}).
\]
To state the random-environment conclusion precisely, take any subsequence;
from it extract a further subsequence on which \eqref{eq:i5-varlimit} and
\eqref{eq:i5-lyap} hold almost surely.  The deterministic Lindeberg--Feller
theorem in almost every conditional environment, followed by the subsequence
principle, gives for every real $u$
\begin{equation}
 \mathbb E\!\left[\exp\{iu n^{-1/2}\sum_IX_{I,n}(\lambda)\}
       \mid\mathcal P_{\ell_n}\right]
 \longrightarrow_{\mathbb P}\exp\{-\tfrac12u^2v_\lambda(t)\}.   \tag{5.16}\label{eq:i5-condcf}
\end{equation}
The zero directions have the same formula by conditional Chebyshev.  Bounded
characteristic functions permit deconditioning, and Cram\'er--Wold proves
\begin{equation}
 {1\over\sqrt n}\sum_I\mathcal Z_I\Longrightarrow N_2(0,B_Q(t)). \tag{5.17}\label{eq:i5-blockclt}
\end{equation}

We finally record every bridge needed to interpret this block theorem.  The
exact upper discrepancy term is $R_{n,\ell_n}$ in \eqref{eq:halfres-d}.
Iteration of \eqref{eq:delta-rec-d}, with one-child chains retained, gives
pathwise
\begin{align}
 \Delta_{Q,n}-\mathbb E\Delta_{Q,n}
 ={}&\sum_I\{\Delta_{Q,N_I}^{I}-a(N_I)\}
   +\{R_{n,\ell_n}-\mathbb ER_{n,\ell_n}\}\notag\\
 &+\left\{\sum_Ia(N_I)-\mathbb E\sum_Ia(N_I)\right\}.       \tag{5.18}\label{eq:i5-delta-cut}
\end{align}
The top term need not be $\mathcal P_{\ell_n}$-measurable and is not
inserted into the conditionally independent array.  Instead,
\eqref{eq:halfres-bound-d} gives
$\mathbb E|R_{n,\ell_n}-\mathbb ER_{n,\ell_n}|^2=O(K_n)=o(n)$.
For the discrepancy block means, a one-label replacement changes
$\sum_Ia(N_I)$ by $z_w-z_{u-1}$, where $z_j=a(j+1)-a(j)$.  On the central
occupancy event, \eqref{eq:zreg-d} bounds the squared change by
$C\lambda_n^{1-2\beta}$ after averaging $|u-w|^2=O(\lambda_n)$; bounded
$z_j$ and Chernoff handle its complement.  Efron--Stein therefore gives,
for any $\beta\in(1/2,1)$,
\begin{equation}
 \operatorname {Var}\!\left(\sum_Ia(N_I)\right)
 \le C_Q n\lambda_n^{1-2\beta}=o(n).                   \tag{5.19}\label{eq:i5-amean}
\end{equation}
The stopped occupancy mean has $L^2$ norm $O(\sqrt {K_n})=o(\sqrt n)$ by
\eqref{eq:actual-occ}.  On $E_{n,\ell_n}$ the stopped identity and
\eqref{eq:joint-cut} give exactly
\[
 V_{2/Q,n}=Qn\ell_n+\sum_IW_{N_I}^{I}+A_n,
 \qquad A_n=\sum_{i=1}^ng_Q(X_i).
\]
The upper entrance projection is
$M_{n,\ell_n}=\sum_i\mathbb E(g_Q\mid X_i(1),\ldots,X_i(\ell_n))$;
its omitted tail has variance $n\|g_Q-g_{Q,\ell_n}\|_2^2=o(n)$.
It is measurable at the cut and has variance $n\sigma_Q^2+o(n)$; the
conditional block characteristic function \eqref{eq:i5-condcf} is
deterministic in the limit, so the standard stable-conditioning argument
combines it with the iid prefix CLT.  The exceptional event satisfies
$\mathbb P(E_{n,\ell_n}^c)\le n(1-K_n^{-1})^{n-1}$; the fourth moments above
and Cauchy--Schwarz make its deletion $o_{L^2}(\sqrt n)$.
Consequently the top-tree, occupancy-centering, truncation-tail, and
singleton residuals are all $o_{\mathbb P}(\sqrt n)$ (indeed the first two

\begin{equation}
 {1\over\sqrt n}\binom{V_{2/Q,n}-\mathbb EV_{2/Q,n}}
 {\Delta_{Q,n}-\mathbb E\Delta_{Q,n}}
 \Longrightarrow N_2\!\left(0,
 \begin{pmatrix}Q^2v_L(t)+2\sigma_Q^2&\kappa_Q(t)\\
 \kappa_Q(t)&\gamma_Q(t)\end{pmatrix}\right).          \tag{5.20}\label{eq:jointphase-d}
\end{equation}

\subsubsection{Parity-specific quantitative nondegeneracy}
We give the insulation argument node by node; in particular, the word

\begin{lemma}[exact guarded-block insulation]\label{lem:exact-insulation-d}
Suppose that three sample points lie in $H$, no other sample point lies in
$I$, and the following guard cells are empty.
\begin{itemize}
\item If $Q=2k+1$, the $k$ depth-$d$ cells immediately to each side of $I$
are empty.
\item If $Q=2k$, the $k$ depth-$d$ cells immediately to each side of $I$
are empty.
\end{itemize}
Condition on every point outside $H$.  If the three coordinates in $H$ are
changed arbitrarily (while remaining distinct and in $H$), then every
Barnes--Hut accept/open decision having a target outside the guarded run is
unchanged; and, for a target in $H$, every decision at a source cell not
contained in $I$ is unchanged.  Source cells containing $I$ are always
opened when their target is in $H$.  Consequently the only variable
directed interactions have both target and source in $I$.  The assertion
continues to hold simultaneously for disjoint guarded runs.
\end{lemma}
\begin{proof}
Dyadic cells are either nested or disjoint, so the cases considered below
exhaust every source node, including every node on a one-child chain.

First let $Q=2k+1$ and let
$R=[u-kw,u+(k+1)w)$ be the guarded run.  For a sample target $y\notin R$,
the distance from $y$ to the midpoint of $I$ is at least
$(k+1/2)w=Qw/2$.  Thus the occupied nonsingleton $I$ is accepted, including
at equality.  Any source ancestor containing both $I$ and $y$ is opened
because it contains the target; any disjoint ancestor has unchanged width,
midpoint, occupancy and multiplicity.  If its traversal reaches the child
on the path to $I$, it eventually accepts $I$, and hence never observes a
descendant of $I$.  This proves the assertion for outside targets, whether
they lie in a sibling cell, in the nearest neighboring ancestor branch, or
arbitrarily far away.

Now take $x\in H$ and a disjoint dyadic source $S=[a,a+s)$ to the right of
$I$.  Since $\theta=2/(2k+1)$, its acceptance inequality is exactly
\[
 {s\over a+s/2-x}\le {2\over2k+1}
 \quad\Longleftrightarrow\quad x\le a-ks.              \tag{P.1}
\]
Equality is on the accepting side.  If $s\ge w$, dyadic alignment gives
$s=2^rw$ and $a-ks$ on the $w$-grid.  It therefore cannot lie in the open
interior of $I$; if it is an endpoint, every $x\in H$ is strictly on the
same side.  If $s<w$, then $s\le w/2$ and guard emptiness forces
$a\ge u+(k+1)w$, whence
$a-ks\ge u+(k/2+1)w>u+w$.  Reflection proves the assertion for left
sources.  These inequalities apply separately at every visited exterior
node, regardless of whether it is empty, a singleton, a nonsingleton, or
part of a one-child chain.  Empty/singleton status is fixed by the exterior
conditioning.  A source ancestor containing $I$ contains $x$ and is
therefore obligatorily opened; its sibling is one of the exterior sources
just treated.

For $Q=2k$, an outside target has distance at least $(k+1/2)w>kw$ from the
midpoint of $I$, so it strictly accepts $I$.  For $x\in H$ and a right
source $S=[a,a+s)$, the exact switch is
\[
 {s\over a+s/2-x}\le {1\over k}
 \quad\Longleftrightarrow\quad x\le a-(k-1/2)s.        \tag{P.2}
\]
If $s\ge2w$, its switch is on the $w$-grid.  It may equal $u$ or $u+w$,
but cannot enter the open interior, and all $x\in H$ are on the same side;
this explicitly covers the equality convention.  If $s=w$, the right guard
gives $a\ge u+(k+1)w$, so the switch is at least $u+3w/2$.  If $s<w$, then
$s\le w/2$ and the switch is at least
$u+(k/2+5/4)w>u+w$.  Reflection gives the left-source inequalities.  The
same ancestor, singleton, empty-node and one-child arguments used above now
apply verbatim.

Finally, a different guarded block is exterior to this one.  The conclusions
just proved say that changing this triple changes neither the traversal into
that block nor any decision made by a target in that block.  Thus the
statement holds for all blocks at once.
\end{proof}

\section{Conditional physical-cell profiles}
For $I=[a,a+s)$ put
\begin{equation}\label{eq:profile}
p_I=\int_I f(x)\,dx,\qquad f_I(u)={s f(a+su)\over p_I},\quad 0\le u\le1.
\end{equation}
These are physical dyadic cells; no quantile change of coordinates is made.

\begin{theorem}[Profile factorization and quantitative flatness]\label{thm:profiles}
For every dyadic $I$,
\begin{align}
 &\int_0^1f_I=1,\qquad m_fs\le p_I\le M_fs,\label{eq:profile-basic}\\
 &\norm{f_I-1}_\infty\le {L_f s\over2m_f},\qquad
 \norm{f_I-1}_1,\norm{f_I-1}_2\le {L_fs\over2m_f}.
 \label{eq:flat}
\end{align}
Conditional on $N_I=k$ and on the identities of the labels in $I$, their
affine coordinates are iid with density $f_I$.  Conditional on the complete
label allocation among all cells at a fixed depth, the coordinate families
in distinct cells are independent.  If $I_0,I_1$ are the children of $I$,
then the next bit has parameter
\begin{equation}\label{eq:q}
q_I={p_{I_1}\over p_I},\qquad |q_I-\tfrac12|\le {L_fs\over8m_f},
\end{equation}
and the two child profiles are
\begin{equation}\label{eq:children}
 (f_I)_0(u)={f_I(u/2)\over2(1-q_I)},\qquad
 (f_I)_1(u)={f_I((1+u)/2)\over2q_I}.
\end{equation}
They must not be identified with one another or with the uniform density.
\end{theorem}
\begin{proof}
The substitution $x=a+su$ proves normalization.  The bounds for $p_I$
follow by integrating \eqref{eq:f-ass}.  With
$\bar f_I=s^{-1}\int_I f$, for every $u$,
\[
 |f(a+su)-\bar f_I|
 \le\int_0^1|f(a+su)-f(a+sv)|\,dv
 \le L_fs\int_0^1|u-v|\,dv\le L_fs/2.
\]
Divide by $\bar f_I\ge m_f$; the $L^1,L^2$ estimates follow because the
underlying interval has measure one.

Fix a set $A$ of $k$ labels.  On the event that exactly $A$ lies in $I$, the
joint density contains
$\prod_{i\in A}f(x_i)\1_I(x_i)/p_I$ for these coordinates and a factor
involving only labels outside $A$.  It therefore factorizes.  Affine change
of variables gives \eqref{eq:profile}.  Repeating this calculation for a
complete allocation among disjoint cells gives conditional independence.
The probability of bit one is $p_{I_1}/p_I$, and restriction followed by
affine rescaling gives \eqref{eq:children}.  Finally,
\[
 |p_{I_1}-p_{I_0}|
 =\left|\int_0^{s/2}\{f(a+s/2+v)-f(a+v)\}\,dv\right|
 \le L_fs^2/4.
\]
Since $p_I\ge m_fs$, division by $2p_I$ proves \eqref{eq:q}.
\end{proof}

\begin{remark}
Conditional independence requires the labelled allocation, not merely an
informal assertion that counts are independent.  Fixed-depth counts are
multinomial.  Likewise, after a stopped prefix configuration is fixed, unused
suffixes are independent with their respective profiles $f_I$, not fair-bit
laws.
\end{remark}

\section{External path length}
\begin{theorem}[Expectation and linear variance]\label{thm:path}
Under \eqref{eq:f-ass},
\begin{equation}\label{eq:path-result}
 \E_fL_n=n\log_2n+O_f(n),\qquad \Var_fL_n=O_f(n).
\end{equation}
\end{theorem}
\begin{proof}
A label at $x$ is active at depth $r$ iff one of the other $n-1$ labels lies
in $I_r(x)$.  Tonelli gives the exact formula
\begin{equation}\label{eq:path-exact}
 \E_fL_n=n\sum_{r\ge0}\int_0^1f(x)
 [1-(1-p_r(x))^{n-1}]\,dx,\quad p_r(x)=\int_{I_r(x)}f.
\end{equation}
The Lipschitz estimate gives
$p_r(x)=2^{-r}f(x)+O_f(2^{-2r})$.  Put
$R_x=\lfloor\log_2(nf(x))\rfloor$.  For $r\le R_x-C$ use
$(1-p)^{n-1}\le e^{-(n-1)p}$; the sum of the omissions from $1$ is bounded
by $\sum_{j\ge C}e^{-c2^j}$.  For $r\ge R_x+C$ use
$1-(1-p)^{n-1}\le(n-1)p$; the tail is bounded by a geometric series.
The finitely many middle terms are bounded.  Uniformly in $x$,
\[
 \sum_r[1-(1-p_r(x))^{n-1}]=\log_2(nf(x))+O_f(1).
\]
Insertion in \eqref{eq:path-exact} proves the expectation statement (indeed
there is the additional displayed term $n\int f\log_2 f$).

For variance, let $\phi(k)=k\1_{\{k\ge2\}}$.  Pathwise,
\begin{equation}\label{eq:Loccup}
L(Y)=\sum_{r\ge0}\sum_{I\in\D_r}\phi(N_I(Y)).
\end{equation}
For an old sample $Y$ of size $N$ and a new point $x$, if
$K_r=N_{I_r(x)}(Y)$, the add-one cost equals
\begin{equation}\label{eq:addL}
 A(x;Y)=\sum_{r\ge0}b(K_r),\quad b(0)=0, b(1)=2, b(k)=1\ (k\ge2).
\end{equation}
The coefficient two is the increment $\phi(2)-\phi(1)$.  Set
$R=\lfloor\log_2(N\vee1)\rfloor$.  Termwise comparison with one gives
\begin{equation}\label{eq:Lcenter}
 |A-R|\le\sum_{r<R}\1_{\{K_r\le1\}}+2\sum_{r\ge R}\1_{\{K_r\ge1\}}+C.
\end{equation}
Conditional on $x$, $K_r\sim\mathrm{Bin}(N,p_r(x))$ and
$m_f2^{-r}\le p_r(x)\le M_f2^{-r}$.  For $Np\ge1$,
$\Pp\{K\le1\}\le(1+Np)e^{-cNp}\le C e^{-c' Np}$; for all $Np$,
$\Pp\{K\ge1\}\le\min(1,Np)$.  Minkowski applied to
\eqref{eq:Lcenter} (no independence across depths is asserted) yields
\[
 \norm{A-R}_2\le C_f\sum_{j\ge0}e^{-c_f2^j}
 +C_f\sum_{j\ge0}2^{-j/2}+C_f\le C_f.
\]
Replace $X_k$ by an independent $X_k'$ and let $Y$ be the other points.
Then
$L(X)-L(X^{(k)})=A(X_k;Y)-A(X_k';Y)$; the same deterministic $R$ cancels.
Thus $\E(L-L^{(k)})^2\le C_f$.  Efron--Stein,
$\Var L\le\frac12\sum_k\E(L-L^{(k)})^2$, proves the variance bound.
\end{proof}

\section{Finite Walsh centering under nonfair bits}

 \prod_{j\in S}(1-2B_j(u)).
\end{equation}
Since $Pd^{(m)}=0$, every nonzero set $S$ contains $1$.  Reveal all labelled
bits in lexicographic order (bit time first, label second), stopped strictly
when the label becomes isolated, and let $\F_t$ be the sigma field generated
by revealed prefixes.  Assign each shifted character to its last revealed
active bit.  Every earlier factor and the activity indicator are then

If the terminal bit is $B_t$ in physical cell $I$, put

 \eta_t=(1-2B_t)-(1-2q_I).
\end{equation}
Then $\E(\eta_t\mid\F_{t-1})=0$.  Expanding the terminal sign as
$\eta_t+(1-2q_I)$ gives the exact decomposition
\begin{equation}\label{eq:finite-dec}
 M_n^{(m)}-\E M_n^{(m)}=N_n^{(m)}+(R_n^{(m)}-\E R_n^{(m)}),
\end{equation}
where $N_n^{(m)}=\sum_t C_t\eta_t$ is a martingale and $R_n^{(m)}$ is the

\begin{equation}\label{eq:bracket}
 Q_n^{(m)}=\sum_t4q_{I_t}(1-q_{I_t})C_t^2.

\begin{theorem}[Fixed-window theorem]\label{thm:finite}
For fixed $m$,
\begin{align}
 \Var R_n^{(m)}&=O_{f,m}(n),\label{eq:driftvar}\\
 \E Q_n^{(m)}&=\norm{d^{(m)}}_2^2n\log_2n+O_{f,m}(n),\label{eq:bracketmean}\\
 \Var Q_n^{(m)}&=O_{f,m}(n\log^2(n+1)),\label{eq:bracketvar}\\
 \sum_t\E|C_t\eta_t|^4&=O_{f,m}(n^2\log(n+1)).\label{eq:fourth}
\end{align}
If $\norm{d^{(m)}}_2>0$, the centered $M_n^{(m)}$, normalized by
$\norm{d^{(m)}}_2\sqrt{n\log_2n}$, converges to $N(0,1)$.
\end{theorem}
\begin{proof}[Proof architecture, with all probabilistic reductions exposed]
Replacing one input changes its prefix chain and at each depth at most the
chain of the unique label whose isolation status is deleted or created.
By \eqref{eq:q}, a predictable bias at depth $r$ is $O_{f,m}(2^{-r})$.
The squared $L^2$ influence is therefore bounded by the square of a geometric
series, uniformly in the replacement; Efron--Stein proves
\eqref{eq:driftvar}.

For \eqref{eq:bracketmean}, expand $C_t^2$.  At depth $r$, all terms whose
$m$-bit windows lie away from the terminal activity boundary have, after
conditioning on the physical cell and its count, profile error
$O_{f,m}(2^{-r})$.  There are $n$ label-depth incidences, and summing this
error over $r$ is $O_{f,m}(n)$.  Moving either of the two activity indicators
across the end of an $m$-window changes at most $2m$ incidences per label,
again $O_m(n)$.  On replacing the remaining local profile by Lebesgue
measure, Walsh orthogonality gives $\sum_Sa_S^2=\norm{d^{(m)}}_2^2$.
Consequently the main term is this norm times $\E L_n$, and
Theorem~\ref{thm:path} proves \eqref{eq:bracketmean}.

For concentration, one-input resampling changes only the resampled prefix
and its deletion/insertion companion chains.  If $H$ is their maximum common
prefix length, the bracket influence is $O_{f,m}(H+m)$.  For two iid points,
\[
 \Pp(H\ge h)=\sum_{I\in\D_h}p_I^2\le M_f2^{-h}\sum_Ip_I=M_f2^{-h}.
\]
Thus $\E(H+m)^2=O_{f,m}(\log^2(n+1))$ after truncation at the largest
occupied depth and the exponentially small tail estimate.  Efron--Stein
proves \eqref{eq:bracketvar}.

Conditionally on $\F_{t-1}$, the centered Bernoulli terminal signs are
independent at a fixed bit time and bounded.  Expanding the fourth power,
odd mixed products vanish and yields
\[
 \E((C_t\eta_t)^4\mid\F_{t-1})\le3B_t^2+C_{f,m}B_t,
 \quad B_t=\E((C_t\eta_t)^2\mid\F_{t-1}).
\]
The exact one- and two-label activity calculation gives
$\sum_t\E B_t^2=O_{f,m}(n^2\log(n+1))$ and
$\sum_t\E B_t=O_{f,m}(n\log(n+1))$, proving \eqref{eq:fourth}.
Equations \eqref{eq:bracketmean}--\eqref{eq:bracketvar} imply
$Q/(n\log_2n)\to\norm{d^{(m)}}_2^2$ in probability.  Equation
\eqref{eq:fourth}, divided by $n^2\log_2^2n$, proves conditional Lindeberg.
The martingale triangular-array CLT applies.  Finally
\eqref{eq:driftvar} makes the centered predictable drift negligible by
Chebyshev, proving the assertion for $M_n^{(m)}$.
\end{proof}

\section{Infinite-tail transfer and the all-depth covariance sum}
Put $e_m=d_\theta-d^{(m)}$.  Then $Pe_m=0$, $e_m$ has mean zero on every
cell in $\D_m$, and $\norm{e_m}_2\to0$.

\begin{theorem}[Stopped-energy exhaustion]\label{thm:tail}
There are numbers $b_m\downarrow0$ such that, uniformly in $n\ge2$,
\begin{equation}\label{eq:tailvar}
 \Var_f M_n(e_m)\le b_m n\log(n+1)+O_{f,\theta}(n).
\end{equation}
Consequently, if $c(\theta)>0$,
\begin{equation}\label{eq:canonical}
 \Var_fM_{\theta,n}\sim c(\theta)n\log_2n,\qquad
 {M_{\theta,n}-\E_fM_{\theta,n}\over\sqrt{c(\theta)n\log_2n}}
 \Longrightarrow N(0,1).
\end{equation}
\end{theorem}
\begin{proof}
We record the exhaustive covariance partition, since replacing it by the
phrase ``tail approximation'' would be invalid.  Write
$Y_{i,r}=e_m(T^rX_i)\1_{\{r<\tau_i\}}$.  Then
\begin{equation}\label{eq:covpartition}
 \Var\sum_{i,r}Y_{i,r}
 =\sum_{i,r}\Var Y_{i,r}
 +2\sum_i\sum_{r<s}\Cov(Y_{i,r},Y_{i,s})
 +\sum_{i\ne j}\sum_{r,s}\Cov(Y_{i,r},Y_{j,s}).
\end{equation}
The diagonal sum is at most
$C_f\norm{e_m}_2^2\E L_n$.
For the same-label sum, condition on $I_r(X_i)$.  Lebesgue cancellation
$Pe_m=0$ kills every positive-lag unstopped integral.  Replacing the cell
profile by Lebesgue density costs $C_f2^{-r}$, while stopping at $s$ costs
at most $\min(1,n2^{-s})$.  Cauchy--Schwarz in the affine cell and summation
of
\[
 \sum_{r<s}2^{-r}\min(1,n2^{-s})
\]
show that this part is
$o_m(1)n\log(n+1)+O_f(n)$; the coefficient tends to zero because both
$\norm{e_m}_2$ and the depth-$m$ cell conditional means tend to zero.

For different labels, their two cells are exactly one of: identical at the
shallower depth, strictly nested, or disjoint.  In disjoint cells the exact
fixed-$n$ multinomial kernel is the difference $R-PQ$ of the joint and
product witness probabilities; the mean-value theorem bounds it by the
product of the two cell masses times the mass of their union.  Summing the
resulting scalar matrix by Schur's test uses
$\sum_{I\in\D_r}p_I=1$ and $p_I\asymp_f2^{-r}$.  Conditional centering on
each depth-$(r+m)$ descendant supplies an additional factor $2^{-m}$.

In the nested case, expose the distinguished deeper label.  The covariance
splits into a direct-witness term, controlled by the same scalar cell moment,
and the centered indicator
$\1_{\{X_j\in B\}}-p_B/p_A$.  If relative depth $h\le m$, integration over
the shifted depth-$m$ partition vanishes exactly.  If $h>m$, the Haar
square-sum identity gives, after summing all descendants $B\subset A$,
\begin{equation}\label{eq:haar}
 \sum_{B\subset A,\,\mathrm{depth}(B)-\mathrm{depth}(A)=h}
 \left|\int_B e_m(T^r x)\,dx\right|^2
 \le C\,2^{-h}\norm{e_m\1_A}_2^2.
\end{equation}
The only mixed strip not covered by geometric decay has deeper depth
$s=\lfloor\log_2n\rfloor+O(1)$, $r=s-h$, $h>m$, and its transpose.
Cauchy--Schwarz, \eqref{eq:haar}, and the deep-cell first-moment factor
$2^{-m}$ reduce it to the two occupancy sums
\begin{align*}
 n^2\sum_s2^{-s}e^{-cn2^{-s}}\sum_{r\le s}4^{-r}&=O_f(n),\\
 n\sum_s\min(1,n2^{-s})\sum_{h>m}2^{-h/2}&le
 C_f2^{-m/2}n\log(n+1)+O_f(n).
\end{align*}
These cases exhaust all $(i,j,r,s)$ in \eqref{eq:covpartition}.  The final
coefficient is bounded by a constant times
$\norm{e_m}_2^2+2^{-m}\norm{e_m}_1^2+2^{-m/2}\norm{e_m}_2$, hence tends to
zero.  This proves \eqref{eq:tailvar}.  Apply the fixed-$m$ CLT first, use
\eqref{eq:tailvar} uniformly in $n$, and then let $m\to\infty$; the
converging-together lemma proves \eqref{eq:canonical}.
\end{proof}

\section{The endpoint term}
\begin{theorem}\label{thm:endpoint}
$\Var_fH_{\theta,n}=O_{f,\theta}(n)$ and $\E_f|H_{\theta,n}|=O_{f,\theta}(n)$.
\end{theorem}
\begin{proof}
The entrance variables $h_\theta(X_i)$ are iid and
$\int h_\theta^2f\le M_f\norm{h_\theta}_2^2$.  For terminal variables,
condition on the exact labelled prefix-free family of terminal cells.
Their intervals are disjoint, their affine suffixes are conditionally
independent with densities $f_I\le M_f/m_f$, and hence the conditional
variance sum is at most $C_{f,\theta}n$.  If
$\bar h=\int_0^1h_\theta$, flatness gives
\[
 |\E(h_\theta(T^{\tau_i}X_i)\mid I_i)-\bar h|
 \le C_f|I_i|\norm{h_\theta}_1.
\]
Prefix freeness gives $\sum_i|I_i|\le1$, so the conditional mean of the
terminal sum differs from $n\bar h$ by a bounded quantity.  Total variance
proves the variance assertion.  The first-moment assertion follows directly
from $\E|h(X_i)|\le M_f\norm h_1$ and the analogous conditional terminal
bound.
\end{proof}

\section{Nonuniform stopped-LISTER theorem}
\begin{theorem}\label{thm:stopped}
If $c(\theta)>0$, then
\[
 \Var_fV_{\theta,n}\sim c(\theta)n\log_2n,\qquad
 {V_{\theta,n}-\E_fV_{\theta,n}\over\sqrt{c(\theta)n\log_2n}}
 \Rightarrow N(0,1),
\]
and the same convergence holds with the true standard deviation.
\end{theorem}
\begin{proof}
Combine \eqref{eq:decomp}, Theorems \ref{thm:path}, \ref{thm:tail}, and
\ref{thm:endpoint}.  The centered $L_n$ and $H_n$ have $L^2$ norm
$O(\sqrt n)$ and are negligible on the $\sqrt{n\log n}$ scale.  Their
covariances with $M_n$ are $O(n\sqrt{\log n})$ by Cauchy--Schwarz, hence
$o(n\log n)$.  Slutsky and variance equivalence give both conclusions.
\end{proof}

\section{Root stabilization and adaptive discrepancy}
Let $D(Y)$ denote the root defect in the deterministic coefficient-one
inward-copy identity, with barred convention $D(Y)=0$ for $|Y|<2$.
The guard theorem from the geometric part gives a fixed finite guard,
uniformly for all physical profiles bounded between $m_f/M_f$ and
$M_f/m_f$, and
\begin{equation}\label{eq:Dmom}
 \sup_{I,n}\E_{f_I}|D(X_1,\ldots,X_n)|^p<\infty
 \quad(p<\infty).
\end{equation}
Indeed guard failure is exponentially small in $n$, while off the guard the
deterministic defect is polynomial in $n$; choosing the exponential moment
absorbs every fixed polynomial.

\begin{lemma}[Support inclusion]\label{lem:support}
Let $Y_1,\ldots,Y_n,X$ be iid with any physical-cell profile.  Put
$Z_n=D(Y_1,\ldots,Y_n,X)-D(Y_1,\ldots,Y_n)$.  Let $I^{*j}$ be the $j$-fold
middle half, $S_j=\{X\in I^{*j}\}$, and $G_j$ the event that the old sample
fills the guard for the configuration in $I^{*j}$.  For every $L$,
\begin{equation}\label{eq:support}
 \{Z_n\ne0\}\subset S_L\cup\bigcup_{j<L}(S_j\cap G_j^c).
\end{equation}
\end{lemma}
\begin{proof}
Induct on $L$.  For $L=0$ the right side is the whole space.  Suppose the
claim holds to level $L$.  On $S_L^c$ insertion has already left the inward
copy and cannot affect the root defect when all preceding guards hold.  On
$S_L\cap G_L$, the coefficient-one identity says that both old and inserted
defects equal the corresponding defects after affine transport to the next
middle half; their difference is therefore the next inward increment, which
can be nonzero only on $S_{L+1}$.  The remaining possibility is
$S_L\cap G_L^c$.  Substitution in the induction hypothesis proves the claim
at $L+1$.
\end{proof}

\begin{lemma}[Local add-one stabilization]\label{lem:localstab}
For every $0<\alpha<1$,
\begin{equation}\label{eq:localstab}
 \norm{Z_n}_2\le C_{f,\theta,\alpha}(n+1)^{-\alpha/2}.
\end{equation}
\end{lemma}
\begin{proof}
Density ratios give $\Pp(S_j)\le C_f2^{-j}$.  The fixed guard contains a
fixed number of subcells, each of conditional mass at least $c_f2^{-j}$ in
the original profile, so a union bound gives
$\Pp(G_j^c)\le C_{f,\theta}e^{-c_{f,\theta}n2^{-j}}$.
Here $S_j$ depends only on the new point and $G_j$ only on the old sample;
they are independent.  Letting $L\to\infty$ in \eqref{eq:support},
\[
 \Pp(Z_n\ne0)\le C\sum_{j\ge0}2^{-j}e^{-cn2^{-j}}le {C\over n+1}.
\]
For the last inequality split where $n2^{-j}$ crosses one; on either side
the terms form a geometric sequence after writing them as
$n^{-1}(n2^{-j})e^{-cn2^{-j}}$.  Choose $p>2/(1-\alpha)$ in
\eqref{eq:Dmom}.  H\"older gives
$\E Z_n^2\le(\E|Z_n|^p)^{2/p}\Pp(Z_n\ne0)^{1-2/p}
\le C(n+1)^{-\alpha}$.
\end{proof}

\begin{theorem}[Adaptive discrepancy]\label{thm:delta}
For every fixed $\theta$, $\Var_f\Delta_{\theta,n}=O_{f,\theta}(n)$.
\end{theorem}
\begin{proof}
Insert a point into an old sample of size $m$.  The exact finite-tree identity
expresses its discrepancy add-one cost as
$A^+=\sum_{r\ge0}\delta_r$ along its dyadic ancestor chain.  Conditional on
the depth-$r$ cell and its old occupancy $K_r=k$, $\delta_r$ has the local
law in Lemma~\ref{lem:localstab}.  With $\lambda_r=m2^{-r}$, density bounds
and the binomial estimate
$\E(K+1)^{-\alpha}\le C_\alpha(1+\lambda_r)^{-\alpha}$ give
\[
 \norm{\delta_r}_2\le C(1+\lambda_r)^{-\alpha/2}.
\]
For $\lambda_r\le1$, $\delta_r=0$ when $K_r=0$.  Choose a high moment in
\eqref{eq:Dmom}; H\"older and $\Pp(K_r\ge1)\le C\lambda_r$ give
$\norm{\delta_r}_2\le C\lambda_r^{\alpha/2}$.  Therefore
\[
 \sum_r\norm{\delta_r}_2\le C\sum_r
 \min(\lambda_r,\lambda_r^{-1})^{\alpha/2}\le C.
\]
Apply Minkowski to finite sums; almost-sure isolation makes the pathwise sum
finite, and the displayed domination gives the $L^2$ limit.  Thus
$\sup_m\E|A^+|^2<\infty$.  Replacing one input is one removal followed by
one insertion into the common remaining sample, so its squared influence is
uniformly bounded.  Efron--Stein proves the theorem.
\end{proof}

\section{Mean remainders and the full generic theorem}
\begin{lemma}[Linear first moments]\label{lem:means}
\begin{equation}\label{eq:means}
 |\E_fM_{\theta,n}|+\E_f|H_{\theta,n}|+\E_f|\Delta_{\theta,n}|
 =O_{f,\theta}(n).
\end{equation}
\end{lemma}
\begin{proof}
For $M$, conditional profile flatness and $Pd_\theta=0$ show that the
conditional mean contributed by an active label at depth $r$ is at most
$C_{f,\theta}2^{-r}$ (first truncate $d_\theta$, then use its structured
$L^1$ tail and pass monotonically).  Summing over at most $n$ active labels
and all $r$ gives $|\E M|\le Cn$.  The endpoint assertion was proved in
Theorem~\ref{thm:endpoint}.

For discrepancy, the deterministic tree identity is
$\Delta=\sum_{I:N_I\ge2}D_I$, with one root defect for every occupied
internal cell.  Conditional on $N_I$ and its labelled allocation,
\eqref{eq:Dmom} with $p=1$ gives $\E(|D_I|\mid N_I)\le C$.  Hence
\[
 \E|\Delta|\le C\sum_{r,I}\Pp(N_I\ge2).
\]
Since $N_I\sim\mathrm{Bin}(n,p_I)$,
$\Pp(N_I\ge2)\le\min(1,n(n-1)p_I^2/2)$.  Using
$p_I\le M_f2^{-r}$ and $\sum_{I\in\D_r}p_I=1$ gives at level $r$
\[
 \sum_I\Pp(N_I\ge2)\le\min(2^r,C_fn^22^{-r}).
\]
Splitting at $2^r\asymp n$ and summing the two geometric series gives
$O_f(n)$.  This proves \eqref{eq:means} without using a variance bound as a
substitute for an expectation estimate.
\end{proof}

\begin{theorem}[Full nonuniform generic theorem]\label{thm:full}
Assume \eqref{eq:f-ass} and $c(\theta)>0$ (equivalently, $\theta$ is
nonresonant).  Then
\begin{align*}
 \E_fT_{\theta,n}&={2\over\theta}n\log_2n+O_{f,\theta}(n),\\
 \Var_fT_{\theta,n}&\sim c(\theta)n\log_2n,\\
 {T_{\theta,n}-\E_fT_{\theta,n}\over\sqrt{c(\theta)n\log_2n}}
 &\Rightarrow N(0,1),\qquad
 {T_{\theta,n}-\E_fT_{\theta,n}\over\sqrt{\Var_fT_{\theta,n}}}
 \Rightarrow N(0,1).
\end{align*}
\end{theorem}
\begin{proof}
The mean follows from \eqref{eq:decomp}, Theorem~\ref{thm:path},
Lemma~\ref{lem:means}, and $T=V+\Delta$.  For variance,
Theorems~\ref{thm:stopped} and \ref{thm:delta} give
\[
 |\Cov(V,\Delta)|\le\sqrt{\Var V\Var\Delta}
 =O(n\sqrt{\log n})=o(n\log n).
\]
Thus $\Var T\sim\Var V$.  Moreover
$(\Delta-\E\Delta)/\sqrt{n\log n}\to0$ in $L^2$, so Slutsky gives the first
CLT.  Variance equivalence gives true-variance normalization.
\end{proof}

\section{Explicit open boundary}
No nonuniform resonant theorem is proved here.  At $\theta=2/Q$ the canonical
coefficient vanishes, so every error retained only as $O(n)$ becomes a
leading-order term.  The present tail transfer, endpoint estimate, external
path estimate, and adaptive stabilization do not determine their joint
phase covariance, do not establish a positive phase variance, and do not
supply a resonant true-variance CLT.  These are open obligations rather than
consequences of Theorem~\ref{thm:full}.
